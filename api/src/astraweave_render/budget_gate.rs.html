1//! Command-line gate for `gpu_frame_budget`.
2//!
3//! An enabled gated flag is rejected before the scene is built.
4//! `--name=false` and `--name false` are off and do not bail.
5
6/// True when `name` is enabled on this command line.
7///
8/// Bare `--name` is on, including when the next token is another flag.
9/// `--name=true` is on. The third clause is the `--name <value>` pair:
10/// `true` is on and `false` is off.
11pub fn flag_on(args: &[String], name: &str) -> bool {
12    let equals_true = format!("{name}=true");
13    args.iter().enumerate().any(|(index, arg)| {
14        let next = args.get(index + 1).map(String::as_str);
15        (arg == name && bare_flag(next))
16            || arg == &equals_true
17            || (arg == name && next == Some("true"))
18    })
19}
20
21fn bare_flag(next: Option<&str>) -> bool {
22    match next {
23        None => true,
24        Some(token) if token.starts_with('-') => true,
25        Some(_) => false,
26    }
27}
28
29/// Scaffolding only. No enabled gated flag draws a frame, so no cell can be filled.
30pub fn reject_gated_flags(args: &[String]) -> anyhow::Result<()> {
31    const BLOCKED: &[(&str, &str)] = &[
32        (
33            "--atmosphere",
34            "enabling it skips the HDR and depth clear, so the heavy scene does not land",
35        ),
36        (
37            "--ssgi",
38            "normal and velocity targets are allocated and never written, and ssgi_history is sampled before copy_to_history; the pass writes denoised_view, and record_gated_presentation never copies it onto hdr_tex, so the post pass samples a frame that does not show it",
39        ),
40        (
41            "--taa",
42            "normal and velocity targets are allocated and never written, and taa_history is sampled before copy_to_history; the pass writes resolved_view and sharpened_view, and record_gated_presentation never copies them onto hdr_tex, so the post pass samples a frame that does not show them",
43        ),
44        (
45            "--upscale",
46            "normal and velocity targets are allocated and never written, and upscale_history is sampled before copy_to_history; the pass writes sharpened_view, and record_gated_presentation never copies it onto hdr_tex, so the post pass samples a frame that does not show it",
47        ),
48        (
49            "--dof",
50            "the depth-of-field history target is uncleared and the result is not copied onto HDR",
51        ),
52        (
53            "--clouds",
54            "the raymarch samples cloud_history before writing it, and the composite is not copied onto HDR",
55        ),
56    ];
57    let hits: Vec<String> = BLOCKED
58        .iter()
59        .filter(|(flag, _)| flag_on(args, flag))
60        .map(|(flag, why)| format!("{flag} ({why})"))
61        .collect();
62    if hits.is_empty() {
63        return Ok(());
64    }
65    anyhow::bail!(
66        "gated flags cannot fill a measurement cell: {}",
67        hits.join("; ")
68    )
69}

71#[cfg(test)]
72mod tests {
73    use super::{flag_on, reject_gated_flags};
74
75    fn args(parts: &[&str]) -> Vec<String> {
76        parts.iter().map(|part| (*part).to_string()).collect()
77    }
78
79    fn rejected(parts: &[&str]) -> String {
80        reject_gated_flags(&args(parts))
81            .expect_err("gated flag must fail before a frame")
82            .to_string()
83    }
84
85    #[test]
86    fn reject_gated_flags_covers_each_form() {
87        assert!(reject_gated_flags(&args(&["--scene", "heavy"])).is_ok());
88        for flag in [
89            "--atmosphere",
90            "--ssgi",
91            "--taa",
92            "--upscale",
93            "--dof",
94            "--clouds",
95        ] {
96            let message = rejected(&[flag]);
97            assert!(message.contains(flag), "{message}");
98            assert!(flag_on(&args(&[flag]), flag));
99            assert!(flag_on(&args(&[&format!("{flag}=true")]), flag));
100            assert!(flag_on(&args(&[flag, "true"]), flag));
101            assert!(!flag_on(&args(&[&format!("{flag}=false")]), flag));
102            assert!(!flag_on(&args(&[flag, "false"]), flag));
103            assert!(reject_gated_flags(&args(&[&format!("{flag}=false")])).is_ok());
104            assert!(reject_gated_flags(&args(&[flag, "false"])).is_ok());
105        }
106        let equals = rejected(&["--atmosphere=true"]);
107        assert!(equals.contains("--atmosphere"), "{equals}");
108        let two_token = rejected(&["--atmosphere", "true"]);
109        assert!(two_token.contains("--atmosphere"), "{two_token}");
110        let pair = rejected(&["--atmosphere", "--ssgi"]);
111        assert!(pair.contains("--atmosphere"), "{pair}");
112        assert!(pair.contains("--ssgi"), "{pair}");
113        assert!(pair.contains("ssgi_history"), "{pair}");
114        let ssgi = rejected(&["--ssgi"]);
115        assert!(ssgi.contains("copy_to_history"), "{ssgi}");
116        assert!(ssgi.contains("denoised_view"), "{ssgi}");
117        assert!(ssgi.contains("hdr_tex"), "{ssgi}");
118        assert!(ssgi.contains("record_gated_presentation"), "{ssgi}");
119        let taa = rejected(&["--taa"]);
120        assert!(taa.contains("taa_history"), "{taa}");
121        assert!(taa.contains("resolved_view"), "{taa}");
122        assert!(taa.contains("sharpened_view"), "{taa}");
123        assert!(taa.contains("hdr_tex"), "{taa}");
124        let upscale = rejected(&["--upscale"]);
125        assert!(upscale.contains("upscale_history"), "{upscale}");
126        assert!(upscale.contains("sharpened_view"), "{upscale}");
127        assert!(upscale.contains("hdr_tex"), "{upscale}");
128        let false_then_ssgi = rejected(&["--atmosphere", "false", "--ssgi"]);
129        assert!(
130            !false_then_ssgi.contains("--atmosphere"),
131            "{false_then_ssgi}"
132        );
133        assert!(
134            false_then_ssgi.contains("ssgi_history"),
135            "{false_then_ssgi}"
136        );
137        assert!(rejected(&["--atmosphere", "false", "--atmosphere"]).contains("--atmosphere"));
138    }
139}
