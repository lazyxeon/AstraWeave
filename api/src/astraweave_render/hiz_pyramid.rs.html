1//! Shared Hi-Z (Hierarchical Depth) min-depth pyramid.
2//!
3//! Builds a min-depth mip chain from a depth view. `Renderer::draw_into`
4//! records it only when `sample_hiz` is set, and that build copies the depth
5//! view from the frame that enabled it. Occlusion, SSR, and SSGI do not
6//! sample the pyramid. SSGI reads scene depth.
7//!
8//! The pyramid stores the **minimum** depth in each 2×2 block, so a future
9//! tracer can skip a region whose ray depth is less than that minimum.
10
11/// Min-depth Hi-Z pyramid. Nothing samples [`HizPyramid::view`] until a pass is wired to it.
12pub struct HizPyramid {
13    copy_pipeline: wgpu::ComputePipeline,
14    downsample_pipeline: wgpu::ComputePipeline,
15    /// Downsample layout. The copy layout is depth-only and stays private.
16    bgl: wgpu::BindGroupLayout,
17    #[allow(dead_code)] // texture must be kept alive for views to remain valid
18    hiz_texture: wgpu::Texture,
19    /// Per-mip views for downsample dispatch (single-mip views).
20    #[allow(dead_code)] // views must be kept alive for bind groups to remain valid
21    mip_views: Vec<wgpu::TextureView>,
22    /// Full mip-chain view. No pass samples this until one is wired to it.
23    full_view: wgpu::TextureView,
24    /// bind_groups[0]: copy — src depth → mip 0.
25    /// bind_groups[1..]: downsample — mip[i-1] → mip[i].
26    bind_groups: Vec<wgpu::BindGroup>,
27    mip_count: u32,
28    width: u32,
29    height: u32,
30    /// Depth view bound into the mip-0 copy. A later frame with a different
31    /// view, even at the same size, must build a new pyramid.
32    src_depth: wgpu::TextureView,
33}
34
35impl HizPyramid {
36    /// Create a new Hi-Z pyramid for the given resolution.
37    ///
38    /// `src_depth_view` is a `texture_depth_2d` view (`Depth32Float`). The copy
39    /// stores it into mip 0 as `R32Float`. Downsample reads that copy with
40    /// `textureLoad`. The view is kept so a later frame can tell whether this
41    /// pyramid still copies that source.
42    pub fn new(
43        device: &wgpu::Device,
44        width: u32,
45        height: u32,
46        src_depth_view: &wgpu::TextureView,
47    ) -> Self {
48        let mip_count = (width.max(height) as f32).log2().ceil() as u32;
49        let fmt = wgpu::TextureFormat::R32Float;
50
51        let hiz_texture = device.create_texture(&wgpu::TextureDescriptor {
52            label: Some("hiz_min_pyramid"),
53            size: wgpu::Extent3d {
54                width,
55                height,
56                depth_or_array_layers: 1,
57            },
58            mip_level_count: mip_count,
59            sample_count: 1,
60            dimension: wgpu::TextureDimension::D2,
61            format: fmt,
62            usage: wgpu::TextureUsages::STORAGE_BINDING
63                | wgpu::TextureUsages::TEXTURE_BINDING
64                | wgpu::TextureUsages::COPY_SRC,
65            view_formats: &[],
66        });
67
68        let mip_views: Vec<wgpu::TextureView> = (0..mip_count)
69            .map(|mip| {
70                hiz_texture.create_view(&wgpu::TextureViewDescriptor {
71                    label: Some(&format!("hiz_min_mip_{mip}")),
72                    format: Some(fmt),
73                    dimension: Some(wgpu::TextureViewDimension::D2),
74                    aspect: wgpu::TextureAspect::All,
75                    base_mip_level: mip,
76                    mip_level_count: Some(1),
77                    base_array_layer: 0,
78                    array_layer_count: Some(1),
79                    usage: None,
80                })
81            })
82            .collect();
83
84        let full_view = hiz_texture.create_view(&wgpu::TextureViewDescriptor {
85            label: Some("hiz_min_full"),
86            ..Default::default()
87        });
88
89        let storage_entry = wgpu::BindGroupLayoutEntry {
90            binding: 1,
91            visibility: wgpu::ShaderStages::COMPUTE,
92            ty: wgpu::BindingType::StorageTexture {
93                access: wgpu::StorageTextureAccess::WriteOnly,
94                format: fmt,
95                view_dimension: wgpu::TextureViewDimension::D2,
96            },
97            count: None,
98        };
99        let copy_bgl = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
100            label: Some("hiz_copy_bgl"),
101            entries: &[
102                wgpu::BindGroupLayoutEntry {
103                    binding: 0,
104                    visibility: wgpu::ShaderStages::COMPUTE,
105                    ty: wgpu::BindingType::Texture {
106                        sample_type: wgpu::TextureSampleType::Depth,
107                        view_dimension: wgpu::TextureViewDimension::D2,
108                        multisampled: false,
109                    },
110                    count: None,
111                },
112                storage_entry,
113            ],
114        });
115        let bgl = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
116            label: Some("hiz_pyramid_bgl"),
117            entries: &[
118                wgpu::BindGroupLayoutEntry {
119                    binding: 0,
120                    visibility: wgpu::ShaderStages::COMPUTE,
121                    ty: wgpu::BindingType::Texture {
122                        sample_type: wgpu::TextureSampleType::Float { filterable: false },
123                        view_dimension: wgpu::TextureViewDimension::D2,
124                        multisampled: false,
125                    },
126                    count: None,
127                },
128                storage_entry,
129            ],
130        });
131
132        // Bind group 0: copy — src_depth → hiz mip 0
133        let mut bind_groups = vec![device.create_bind_group(&wgpu::BindGroupDescriptor {
134            label: Some("hiz_copy_bg"),
135            layout: &copy_bgl,
136            entries: &[
137                wgpu::BindGroupEntry {
138                    binding: 0,
139                    resource: wgpu::BindingResource::TextureView(src_depth_view),
140                },
141                wgpu::BindGroupEntry {
142                    binding: 1,
143                    resource: wgpu::BindingResource::TextureView(&mip_views[0]),
144                },
145            ],
146        })];
147
148        // Bind groups 1..N-1: downsample — mip[i] → mip[i+1]
149        for i in 0..mip_count.saturating_sub(1) {
150            bind_groups.push(device.create_bind_group(&wgpu::BindGroupDescriptor {
151                label: Some(&format!("hiz_ds_bg_{i}_{}", i + 1)),
152                layout: &bgl,
153                entries: &[
154                    wgpu::BindGroupEntry {
155                        binding: 0,
156                        resource: wgpu::BindingResource::TextureView(&mip_views[i as usize]),
157                    },
158                    wgpu::BindGroupEntry {
159                        binding: 1,
160                        resource: wgpu::BindingResource::TextureView(&mip_views[(i + 1) as usize]),
161                    },
162                ],
163            }));
164        }
165
166        let copy_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
167            label: Some("hiz_pyramid_copy_shader"),
168            source: wgpu::ShaderSource::Wgsl(
169                include_str!("../shaders/hiz_pyramid_copy.wgsl").into(),
170            ),
171        });
172        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
173            label: Some("hiz_pyramid_shader"),
174            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/hiz_pyramid.wgsl").into()),
175        });
176
177        let copy_pl = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
178            label: Some("hiz_copy_pl"),
179            bind_group_layouts: &[&copy_bgl],
180            push_constant_ranges: &[],
181        });
182        let pl = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
183            label: Some("hiz_pyramid_pl"),
184            bind_group_layouts: &[&bgl],
185            push_constant_ranges: &[],
186        });
187
188        let copy_pipeline = device.create_compute_pipeline(&wgpu::ComputePipelineDescriptor {
189            label: Some("hiz_copy_pipeline"),
190            layout: Some(&copy_pl),
191            module: &copy_shader,
192            entry_point: Some("copy_depth"),
193            compilation_options: Default::default(),
194            cache: None,
195        });
196
197        let downsample_pipeline =
198            device.create_compute_pipeline(&wgpu::ComputePipelineDescriptor {
199                label: Some("hiz_downsample_pipeline"),
200                layout: Some(&pl),
201                module: &shader,
202                entry_point: Some("downsample"),
203                compilation_options: Default::default(),
204                cache: None,
205            });
206
207        Self {
208            copy_pipeline,
209            downsample_pipeline,
210            bgl,
211            hiz_texture,
212            mip_views,
213            full_view,
214            bind_groups,
215            mip_count,
216            width,
217            height,
218            src_depth: src_depth_view.clone(),
219        }
220    }
221
222    /// Full mip-chain view. No pass samples this until one is wired to it.
223    pub fn view(&self) -> &wgpu::TextureView {
224        &self.full_view
225    }
226
227    /// The R32Float pyramid texture. A consumer samples [`Self::view`].
228    pub fn texture(&self) -> &wgpu::Texture {
229        &self.hiz_texture
230    }
231
232    /// The mip-0 copy was bound from this depth view.
233    ///
234    /// wgpu compares view identity, so a second view of the same texture is
235    /// a different source and needs its own pyramid.
236    pub fn copies_from(&self, view: &wgpu::TextureView) -> bool {
237        &self.src_depth == view
238    }
239
240    /// Number of mip levels in the pyramid.
241    pub fn mip_count(&self) -> u32 {
242        self.mip_count
243    }
244
245    /// Resolution of the pyramid (matches source depth).
246    pub fn dimensions(&self) -> (u32, u32) {
247        (self.width, self.height)
248    }
249
250    /// Build the Hi-Z pyramid from the current frame's depth buffer.
251    ///
252    /// 1. Copies the source depth into mip 0 (using the copy compute shader).
253    /// 2. Downsamples each mip level using min-depth reduction.
254    pub fn build(&self, encoder: &mut wgpu::CommandEncoder) {
255        // Pass 0: copy source depth → mip 0
256        {
257            let mut pass = encoder.begin_compute_pass(&wgpu::ComputePassDescriptor {
258                label: Some("hiz_copy_depth"),
259                timestamp_writes: None,
260            });
261            pass.set_pipeline(&self.copy_pipeline);
262            pass.set_bind_group(0, &self.bind_groups[0], &[]);
263            pass.dispatch_workgroups(self.width.div_ceil(8), self.height.div_ceil(8), 1);
264        }
265
266        // Passes 1..N: downsample mip[i] → mip[i+1]
267        for i in 0..self.mip_count.saturating_sub(1) {
268            let mip_w = (self.width >> (i + 1)).max(1);
269            let mip_h = (self.height >> (i + 1)).max(1);
270
271            let mut pass = encoder.begin_compute_pass(&wgpu::ComputePassDescriptor {
272                label: Some("hiz_downsample"),
273                timestamp_writes: None,
274            });
275            pass.set_pipeline(&self.downsample_pipeline);
276            // bind_groups[i+1] maps mip[i] → mip[i+1]
277            pass.set_bind_group(0, &self.bind_groups[(i + 1) as usize], &[]);
278            pass.dispatch_workgroups(mip_w.div_ceil(8), mip_h.div_ceil(8), 1);
279        }
280    }
281
282    /// Recreate the pyramid for a new resolution and/or depth view.
283    pub fn resize(
284        &mut self,
285        device: &wgpu::Device,
286        width: u32,
287        height: u32,
288        src_depth_view: &wgpu::TextureView,
289    ) {
290        if self.width == width && self.height == height && self.copies_from(src_depth_view) {
291            return;
292        }
293        *self = Self::new(device, width, height, src_depth_view);
294    }
295
296    /// Bind group layout (exposed for other systems that need to match).
297    pub fn bind_group_layout(&self) -> &wgpu::BindGroupLayout {
298        &self.bgl
299    }

306    #[test]
307    fn hiz_mip_count_1920x1080() {
308        let mip_count = 1920_f32.log2().ceil() as u32;
309        assert_eq!(mip_count, 11); // 1920 → ceil(log2(1920)) = 11
310    }
311
312    #[test]
313    fn hiz_mip_count_power_of_two() {
314        let mip_count = 1024_f32.log2().ceil() as u32;
315        assert_eq!(mip_count, 10); // 1024 → log2(1024) = 10
316    }

318    #[test]
319    fn hiz_pyramid_creation() {
320        let instance = wgpu::Instance::new(&wgpu::InstanceDescriptor::default());
321        let adapter =
322            match pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
323                power_preference: wgpu::PowerPreference::LowPower,
324                compatible_surface: None,
325                force_fallback_adapter: false,
326            })) {
327                Ok(adapter) => adapter,
328                Err(err) => {
329                    let reason = format!("no wgpu adapter: {err}");
330                    if crate::gpu_capability::skip_if_unavailable() {
331                        crate::gpu_capability::note_skip(&reason);
332                        return;
333                    }
334                    panic!("Failed to find adapter: {reason}");
335                }
336            };
337        let (device, _queue) =
338            pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor::default()))
339                .expect("device");
340        if crate::gpu_fault::skip_without_depth_texture_load(&device, "hiz_pyramid_creation") {
341            return;
342        }
343
344        // Create a dummy R32Float depth source texture
345        let depth_tex = device.create_texture(&wgpu::TextureDescriptor {
346            label: Some("test_depth"),
347            size: wgpu::Extent3d {
348                width: 1920,
349                height: 1080,
350                depth_or_array_layers: 1,
351            },
352            mip_level_count: 1,
353            sample_count: 1,
354            dimension: wgpu::TextureDimension::D2,
355            format: wgpu::TextureFormat::Depth32Float,
356            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::RENDER_ATTACHMENT,
357            view_formats: &[],
358        });
359        let depth_view = depth_tex.create_view(&wgpu::TextureViewDescriptor::default());
360
361        let pyramid = HizPyramid::new(&device, 1920, 1080, &depth_view);
362        assert_eq!(pyramid.dimensions(), (1920, 1080));
363        assert_eq!(pyramid.mip_count(), 11);
364        assert!(pyramid.copies_from(&depth_view));
365        let other = depth_tex.create_view(&wgpu::TextureViewDescriptor::default());
366        assert!(
367            !pyramid.copies_from(&other),
368            "a second view of the same texture is a different copy source"
369        );
370    }
