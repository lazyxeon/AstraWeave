16/// `BlockingPool::drop` joins a stuck `std::fs::read` with no timeout.
17/// Shutdown waits this long, then leaves that read behind so `Drop` can return.
18const DECODE_SHUTDOWN_BOUND: Duration = Duration::from_millis(500);
19
20/// Decoded images waiting to upload. The deque is reserved once in `new`.
21const UPLOAD_RING_CAP: usize = 8;
22
23/// Unique identifier for textures in the asset system
24pub type AssetId = String;

26/// GPU texture handle wrapping the actual resource
27#[derive(Debug, Clone)]
28pub struct TextureHandle {
29    pub id: AssetId,
30    pub texture: Arc<Texture>,
31    pub width: u32,
32    pub height: u32,
33    pub mip_levels: u32,
34    pub memory_bytes: usize,
35}

37/// State of an asset in the streaming system
38#[derive(Debug)]
39enum AssetState {
40    Loading,
41    Resident(TextureHandle),
42    Failed(#[allow(dead_code)] String),
43}
44
45/// Texture load request with priority
46#[derive(Debug, Clone)]
47struct LoadRequest {
48    id: AssetId,
49    priority: u32,
50    distance: f32, // Distance from camera
51}
52
53impl Eq for LoadRequest {}
54
55impl PartialEq for LoadRequest {
56    fn eq(&self, other: &Self) -> bool {
57        self.priority == other.priority && self.id == other.id
58    }
59}
60
61impl PartialOrd for LoadRequest {
62    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
63        Some(self.cmp(other))
64    }
65}
66
67impl Ord for LoadRequest {
68    fn cmp(&self, other: &Self) -> Ordering {
69        // Higher priority first, then closer distance
70        match self.priority.cmp(&other.priority) {
71            Ordering::Equal => {
72                // Reverse distance comparison (closer is higher priority)
73                other
74                    .distance
75                    .partial_cmp(&self.distance)
76                    .unwrap_or(Ordering::Equal)
77            }
78            other => other,
79        }
80    }
81}

83type LoadResult = Result<(AssetId, Texture), (AssetId, String)>;

85enum StreamMsg {
86    Ready {
87        id: AssetId,
88        width: u32,
89        height: u32,
90        rgba: Vec<u8>,
91    },
92    Failed {
93        id: AssetId,
94        error: String,
95    },

98impl StreamMsg {
99    fn id(&self) -> &str {
100        match self {
101            Self::Ready { id, .. } | Self::Failed { id, .. } => id,
102        }
103    }

106/// Texture streaming manager with LRU eviction and priority-based loading
107pub struct TextureStreamingManager {
108    /// Map of asset states (Resident, Loading, Failed)
109    assets: HashMap<AssetId, AssetState>,
110    /// LRU queue for eviction (front = oldest)
111    lru_queue: VecDeque<AssetId>,
112    /// Priority queue for pending load requests
113    load_queue: BinaryHeap<LoadRequest>,
114    /// Set of IDs currently in load_queue, for O(1) dedup
115    pending_ids: HashSet<AssetId>,
116    /// Maximum GPU memory budget in bytes
117    max_memory_bytes: usize,
118    /// Current GPU memory usage in bytes
119    current_memory_bytes: usize,
120    /// Camera position for distance-based residency
121    camera_position: Vec3,
122
123    /// Channel for receiving async load results (`process_next_load`).
124    result_rx: tokio_mpsc::Receiver<LoadResult>,
125    /// Sender to clone for async tasks (`process_next_load`).
126    result_tx: tokio_mpsc::Sender<LoadResult>,
127    /// Decode workers. The runtime lives on its own thread so dropping this
128    /// manager from inside another Tokio runtime does not block.
129    runtime: DecodeRuntime,
130    /// Pre-sized ring of decoded images. Uploads drain it on the frame.
131    upload_ring: VecDeque<StreamMsg>,
132    decoded_tx: SyncSender<StreamMsg>,
133    decoded_rx: Receiver<StreamMsg>,
134}

136/// One current-thread runtime on `texture-stream`. Drop signals that thread
137/// and only joins it when the caller is not already inside a runtime.
138struct DecodeRuntime {
139    handle: tokio::runtime::Handle,
140    shutdown: Option<tokio::sync::oneshot::Sender<()>>,
141    thread: Option<std::thread::JoinHandle<()>>,
142}
143
144impl DecodeRuntime {
145    fn start() -> anyhow::Result<Self> {
146        let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();
147        let (ready_tx, ready_rx) = std::sync::mpsc::channel();
148        let thread = std::thread::Builder::new()
149            .name("texture-stream".into())
150            .spawn(move || {
151                let runtime = tokio::runtime::Builder::new_current_thread()
152                    .enable_all()
153                    .build();
154                let runtime = match runtime {
155                    Ok(runtime) => runtime,
156                    Err(err) => {
157                        let _ = ready_tx.send(Err(err));
158                        return;
159                    }
160                };
161                let handle = runtime.handle().clone();
162                if ready_tx.send(Ok(handle)).is_err() {
163                    runtime.shutdown_timeout(DECODE_SHUTDOWN_BOUND);
164                    return;
165                }
166                // A current-thread runtime polls `Handle::spawn` tasks only
167                // while this thread is inside `block_on`. Waiting on a std
168                // channel left every decode queued, so the fifo reader in
169                // `stream_fallback` never opened.
170                runtime.block_on(async move {
171                    let _ = shutdown_rx.await;
172                });
173                // Dropping the runtime joins blocking threads forever. A fifo
174                // read in `stream_fallback` is still in `open` when an assert
175                // fails before the writer is released.
176                runtime.shutdown_timeout(DECODE_SHUTDOWN_BOUND);
177            })
178            .context("texture streaming runtime")?;
179        let handle = ready_rx
180            .recv()
181            .context("texture streaming runtime stopped")?
182            .context("texture streaming runtime")?;
183        Ok(Self {
184            handle,
185            shutdown: Some(shutdown_tx),
186            thread: Some(thread),
187        })
188    }
189
190    fn spawn<F>(&self, future: F)
191    where
192        F: std::future::Future<Output = ()> + Send + 'static,
193    {
194        self.handle.spawn(future);
195    }
196}
197
198impl Drop for DecodeRuntime {
199    fn drop(&mut self) {
200        self.shutdown.take();
201        let Some(thread) = self.thread.take() else {
202            return;
203        };
204        if tokio::runtime::Handle::try_current().is_err() {
205            let _ = thread.join();
206        }
207    }
208}

210impl TextureStreamingManager {
211    /// Create a new texture streaming manager.
212    ///
213    /// The decode runtime is one tokio worker. If that runtime cannot start,
214    /// the error is returned and the renderer does not continue without it.
215    ///
216    /// # Arguments
217    /// * `max_memory_mb` - Maximum GPU memory budget in megabytes
218    pub fn new(max_memory_mb: usize) -> anyhow::Result<Self> {
219        let (tx, rx) = tokio_mpsc::channel(32); // Buffer size of 32 results
220        let (decoded_tx, decoded_rx) = mpsc::sync_channel(UPLOAD_RING_CAP);
221        let mut upload_ring = VecDeque::new();
222        upload_ring.reserve(UPLOAD_RING_CAP);
223        let runtime = DecodeRuntime::start()?;
224        Ok(Self {
225            assets: HashMap::new(),
226            lru_queue: VecDeque::new(),
227            load_queue: BinaryHeap::new(),
228            pending_ids: HashSet::new(),
229            max_memory_bytes: max_memory_mb * 1024 * 1024,
230            current_memory_bytes: 0,
231            camera_position: Vec3::ZERO,
232            result_rx: rx,
233            result_tx: tx,
234            runtime,
235            upload_ring,
236            decoded_tx,
237            decoded_rx,
238        })
239    }

241    /// Queue `id` by distance from the current camera. Closer requests sort first.
242    pub fn request_at(&mut self, id: impl Into<String>, world_pos: Vec3) -> Option<TextureHandle> {
243        let id = id.into();
244        let distance = world_pos.distance(self.camera_position);
245        let priority = if distance.is_finite() {
246            (1_000_000.0 / (1.0 + distance)) as u32
247        } else {
248            0
249        };
250        self.request_texture(id, priority, distance)
251    }

254        self.camera_position = camera;

257    /// Pull decoded images into the ring and upload at most `max_uploads`.
258    /// Also starts at most `max_uploads` decode workers. An empty queue does
259    /// not allocate. Load failures stay on the asset as `Failed`; the previous
260    /// resident texture, or the caller's 1×1 fallback, remains the bind.
261    pub fn tick(&mut self, device: &wgpu::Device, queue: &wgpu::Queue, max_uploads: usize) {
262        while self.upload_ring.len() < UPLOAD_RING_CAP {
263            match self.decoded_rx.try_recv() {
264                Ok(msg) => self.upload_ring.push_back(msg),
265                Err(_) => break,
266            }
267        }
268        let mut uploaded = 0;
269        while uploaded < max_uploads {
270            let Some(msg) = self.upload_ring.pop_front() else {
271                break;
272            };
273            self.admit(device, queue, msg);
274            uploaded += 1;
275        }
276        let mut started = 0;
277        while started < max_uploads {
278            if !self.start_one_decode() {
279                break;
280            }
281            started += 1;
282        }
283    }

299    fn start_one_decode(&mut self) -> bool {
300        let Some(request) = self.load_queue.pop() else {
301            return false;
302        };
303        self.pending_ids.remove(&request.id);
304        if self.assets.contains_key(&request.id) {
305            return true;
306        }
307        self.assets.insert(request.id.clone(), AssetState::Loading);
308        let tx = self.decoded_tx.clone();
309        let id = request.id.clone();
310        self.runtime.spawn(async move {
311            let msg = decode_texture(id).await;
312            let label = msg.id().to_string();
313            let send = tokio::task::spawn_blocking(move || tx.send(msg)).await;
314            match send {
315                Ok(Ok(())) => {}
316                Ok(Err(err)) => {
317                    error!("texture stream closed before {} was admitted", err.0.id());
318                }
319                Err(err) => {
320                    error!("texture stream worker stopped for {label}: {err}");
321                }
322            }
323        });
324        true
325    }

327    fn admit(&mut self, device: &wgpu::Device, queue: &wgpu::Queue, msg: StreamMsg) {
328        match msg {
329            StreamMsg::Failed { id, error } => {
330                error!("Texture load failed for {id}: {error}");
331                self.assets.insert(id, AssetState::Failed(error));
332            }
333            StreamMsg::Ready {
334                id,
335                width,
336                height,
337                rgba,
338            } => {
339                let memory_bytes =
340                    (width as usize) * (height as usize) * 4 + (width as usize * height as usize);
341                while self.current_memory_bytes + memory_bytes > self.max_memory_bytes {
342                    if !self.evict_lru() {
343                        error!("Memory budget full, cannot admit {id}");
344                        self.assets
345                            .insert(id, AssetState::Failed("memory budget full".into()));
346                        return;
347                    }
348                }
349                let texture = device.create_texture(&wgpu::TextureDescriptor {
350                    label: Some("streamed albedo"),
351                    size: wgpu::Extent3d {
352                        width,
353                        height,
354                        depth_or_array_layers: 1,
355                    },
356                    mip_level_count: 1,
357                    sample_count: 1,
358                    dimension: wgpu::TextureDimension::D2,
359                    format: wgpu::TextureFormat::Rgba8UnormSrgb,
360                    usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
361                    view_formats: &[],
362                });
363                queue.write_texture(
364                    wgpu::TexelCopyTextureInfo {
365                        texture: &texture,
366                        mip_level: 0,
367                        origin: wgpu::Origin3d::ZERO,
368                        aspect: wgpu::TextureAspect::All,
369                    },
370                    &rgba,
371                    wgpu::TexelCopyBufferLayout {
372                        offset: 0,
373                        bytes_per_row: Some(4 * width),
374                        rows_per_image: Some(height),
375                    },
376                    wgpu::Extent3d {
377                        width,
378                        height,
379                        depth_or_array_layers: 1,
380                    },
381                );
382                let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
383                let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
384                    label: Some("streamed albedo sampler"),
385                    mag_filter: wgpu::FilterMode::Linear,
386                    min_filter: wgpu::FilterMode::Linear,
387                    ..Default::default()
388                });
389                let handle = TextureHandle {
390                    id: id.clone(),
391                    texture: Arc::new(Texture {
392                        texture,
393                        view,
394                        sampler,
395                    }),
396                    width,
397                    height,
398                    mip_levels: 1,
399                    memory_bytes,
400                };
401                self.current_memory_bytes += memory_bytes;
402                self.assets.insert(id.clone(), AssetState::Resident(handle));
403                self.lru_queue.push_back(id);
404            }
405        }
406    }

408    /// Request a texture with given priority
409    ///
410    /// Returns immediately if texture is already loaded, otherwise queues for async loading.
411    ///
412    /// # Arguments
413    /// * `id` - Asset ID of the texture (assumed to be file path)
414    /// * `priority` - Load priority (higher = more urgent)
415    /// * `distance` - Distance from camera (for tie-breaking)
416    ///
417    /// # Returns
418    /// * `Some(TextureHandle)` if texture is resident in GPU memory
419    /// * `None` if texture needs to be loaded (queued)
420    pub fn request_texture(
421        &mut self,
422        id: AssetId,
423        priority: u32,
424        distance: f32,
425    ) -> Option<TextureHandle> {
426        // Check if resident
427        if let Some(AssetState::Resident(handle)) = self.assets.get(&id) {
428            // Update LRU (move to back)
429            if let Some(pos) = self.lru_queue.iter().position(|x| x == &id) {
430                self.lru_queue.remove(pos);
431                self.lru_queue.push_back(id.clone());
432            }
433            return Some(handle.clone());
434        }
435
436        // Check if already loading or failed — don't re-queue
437        if self.assets.contains_key(&id) {
438            return None;
439        }
440
441        // Check if already pending in the load queue — don't double-queue
442        if self.pending_ids.contains(&id) {
443            return None;
444        }
445
446        // Queue for load — do NOT insert into assets map yet.
447        // The Loading state will be set by process_next_load when it actually
448        // picks this request off the queue and spawns the async task.
449        self.pending_ids.insert(id.clone());
450        self.load_queue.push(LoadRequest {
451            id,
452            priority,
453            distance,
454        });
455
456        None
457    }

459    /// Process the next load request from the queue
460    ///
461    /// # Arguments
462    /// * `device` - WGPU device for loading
463    /// * `queue` - WGPU queue for loading
464    pub fn process_next_load(&mut self, device: &Arc<wgpu::Device>, queue: &Arc<wgpu::Queue>) {
465        // 1. Process completed loads
466        while let Ok(result) = self.result_rx.try_recv() {
467            match result {
468                Ok((id, texture)) => {
469                    let width = texture.texture.size().width;
470                    let height = texture.texture.size().height;
471                    let mip_levels = texture.texture.mip_level_count();
472                    // Approximate memory: width * height * 4 bytes * 1.33 for mips
473                    let memory_bytes =
474                        (width * height * 4) as usize + ((width * height * 4) as usize / 3);
475
476                    // Check budget before committing
477                    let mut admitted = true;
478                    while self.current_memory_bytes + memory_bytes > self.max_memory_bytes {
479                        if !self.evict_lru() {
480                            error!("Memory budget full, cannot admit {id}");
481                            self.assets.insert(
482                                id.clone(),
483                                AssetState::Failed("memory budget full".into()),
484                            );
485                            admitted = false;
486                            break;
487                        }
488                    }
489                    if !admitted {
490                        continue;
491                    }
492
493                    let handle = TextureHandle {
494                        id: id.clone(),
495                        texture: Arc::new(texture),
496                        width,
497                        height,
498                        mip_levels,
499                        memory_bytes,
500                    };
501
502                    self.current_memory_bytes += memory_bytes;
503                    self.assets.insert(id.clone(), AssetState::Resident(handle));
504                    self.lru_queue.push_back(id.clone());
505                    debug!(
506                        "Texture loaded: {} ({} MB)",
507                        id,
508                        memory_bytes as f32 / 1024.0 / 1024.0
509                    );
510                }
511                Err((id, err)) => {
512                    error!("Texture load failed for {}: {}", id, err);
513                    self.assets.insert(id, AssetState::Failed(err));
514                }
515            }
516        }
517
518        // 2. Start new loads from queue
519        // Pop one per call to throttle
520        if let Some(request) = self.load_queue.pop() {
521            // Remove from pending set since we're processing it now
522            self.pending_ids.remove(&request.id);
523
524            // Skip if already resident, loading, or failed
525            if self.assets.contains_key(&request.id) {
526                return;
527            }
528
529            // Now mark as Loading — this is the single insertion point for Loading state
530            self.assets.insert(request.id.clone(), AssetState::Loading);
531
532            // Spawn async task
533            let device = device.clone();
534            let queue = queue.clone();
535            let tx = self.result_tx.clone();
536            let id = request.id.clone();
537            let path = request.id.clone(); // Assuming ID is path
538
539            self.runtime.spawn(async move {
540                let result =
541                    Texture::load_texture_async(&device, &queue, &path, TextureUsage::Albedo).await;
542
543                match result {
544                    Ok(texture) => {
545                        if tx.send(Ok((id, texture))).await.is_err() {
546                            error!("texture stream closed before {path} was admitted");
547                        }
548                    }
549                    Err(e) => {
550                        let message = e.to_string();
551                        if tx.send(Err((id, message.clone()))).await.is_err() {
552                            error!(
553                                "texture stream closed, load failure for {path} was not recorded: {message}"
554                            );
555                        }
556                    }
557                }
558            });
559        }
560    }

562    /// Evict the least recently used texture
563    ///
564    /// # Returns
565    /// * `true` if a texture was evicted
566    /// * `false` if no textures to evict
567    pub fn evict_lru(&mut self) -> bool {
568        if let Some(id) = self.lru_queue.pop_front() {
569            if let Some(AssetState::Resident(handle)) = self.assets.remove(&id) {
570                self.current_memory_bytes = self
571                    .current_memory_bytes
572                    .saturating_sub(handle.memory_bytes);
573                debug!(
574                    "Evicted texture {} ({:.2}MB)",
575                    id,
576                    handle.memory_bytes as f32 / (1024.0 * 1024.0)
577                );
578                return true;
579            }
580        }
581        false
582    }

599    /// Get current memory usage statistics
600    pub fn get_stats(&self) -> TextureStreamingStats {
601        let loaded_count = self
602            .assets
603            .values()
604            .filter(|s| matches!(s, AssetState::Resident(_)))
605            .count();
606        let pending_count = self.load_queue.len()
607            + self
608                .assets
609                .values()
610                .filter(|s| matches!(s, AssetState::Loading))
611                .count();
612
613        TextureStreamingStats {
614            loaded_count,
615            pending_count,
616            memory_used_bytes: self.current_memory_bytes,
617            memory_budget_bytes: self.max_memory_bytes,
618            memory_used_percent: (self.current_memory_bytes as f32 / self.max_memory_bytes as f32)
619                * 100.0,
620        }
621    }

639async fn decode_texture(id: String) -> StreamMsg {
640    let bytes = match crate::texture::read_capped_image(&id).await {
641        Ok(bytes) => bytes,
642        Err(error) => {
643            return StreamMsg::Failed { id, error };
644        }
645    };
646    let decoded = tokio::task::spawn_blocking(move || decode_bytes(&bytes)).await;
647    match decoded {
648        Ok(Ok((width, height, rgba))) => StreamMsg::Ready {
649            id,
650            width,
651            height,
652            rgba,
653        },
654        Ok(Err(err)) => StreamMsg::Failed {
655            id,
656            error: err.to_string(),
657        },
658        Err(err) => StreamMsg::Failed {
659            id,
660            error: format!("decode worker stopped: {err}"),
661        },
662    }
663}

665fn decode_bytes(bytes: &[u8]) -> anyhow::Result<(u32, u32, Vec<u8>)> {
666    #[cfg(feature = "textures")]
667    {
668        let image = astraweave_asset::decode_limited_image(bytes)?;
669        Ok((image.width(), image.height(), image.to_rgba8().into_raw()))
670    }
671    #[cfg(not(feature = "textures"))]
672    {
673        let _ = bytes;
674        anyhow::bail!("textures feature is off");
675    }
676}

1050    fn fifo_at_the_cap_does_not_read_one_more_byte() {
1051        let dir = tempfile::tempdir().expect("tempdir");
1052        let path = dir.path().join("exact.fifo");
1053        let made = std::process::Command::new("mkfifo")
1054            .arg(&path)
1055            .status()
1056            .expect("mkfifo");
1057        assert!(made.success(), "mkfifo");
1058        let cap = astraweave_asset::image_limits::MAX_IMAGE_ALLOC;
1059        let path_for_writer = path.clone();
1060        let (stop_tx, stop_rx) = std::sync::mpsc::channel::<()>();
1061        let writer = std::thread::spawn(move || {
1062            use std::io::Write;
1063            let mut file = std::fs::OpenOptions::new()
1064                .write(true)
1065                .open(&path_for_writer)
1066                .expect("fifo writer");
1067            let chunk = vec![0u8; 1024 * 1024];
1068            let mut left = cap;
1069            while left > 0 {
1070                let n = std::cmp::min(left, chunk.len() as u64) as usize;
1071                if file.write_all(&chunk[..n]).is_err() {
1072                    return;
1073                }
1074                left -= n as u64;
1075            }
1076            let _ = stop_rx.recv();
1077        });
1078        let runtime = tokio::runtime::Builder::new_current_thread()
1079            .enable_all()
1080            .build()
1081            .expect("runtime");
1082        let path_str = path.to_string_lossy().into_owned();
1083        let started = std::time::Instant::now();
1084        let msg = runtime.block_on(crate::texture::read_capped_image(&path_str));
1085        let elapsed = started.elapsed();
1086        let _ = stop_tx.send(());
1087        let _ = writer.join();
1088        match msg {
1089            Err(error) => assert!(error.contains(&cap.to_string()), "{error}"),
1090            Ok(bytes) => panic!("read {} bytes and waited for more", bytes.len()),
1091        }
1092        assert!(
1093            elapsed < Duration::from_secs(30),
1094            "read blocked past the cap for {elapsed:?}"
1095        );
1096    }

1099    fn unblock_fifo_writer(path: &std::path::Path) {
1100        use std::os::unix::fs::OpenOptionsExt;
1101        // Linux and Android use 0x800. The other Unix targets use 0x4.
1102        #[cfg(any(target_os = "linux", target_os = "android"))]
1103        const O_NONBLOCK: i32 = 0x800;
1104        #[cfg(not(any(target_os = "linux", target_os = "android")))]
1105        const O_NONBLOCK: i32 = 0x4;
1106        let _ = std::fs::OpenOptions::new()
1107            .write(true)
1108            .custom_flags(O_NONBLOCK)
1109            .open(path);
1110    }
