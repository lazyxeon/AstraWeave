1//! GPU timestamp profiler for per-pass performance analysis.
2//!
3//! Wraps wgpu's timestamp query mechanism to automatically capture timing for
4//! each render/compute pass. Results are read back asynchronously and exposed
5//! as per-pass millisecond durations.
6//!
7//! Requires `Features::TIMESTAMP_QUERY` on the device.
8
9use std::collections::BTreeMap;
10use std::sync::{
11    atomic::{AtomicBool, Ordering},
12    Arc,
13};

15/// Copied at the end of a frame encoder. A dropped command buffer leaves the
16/// readback at 0.
17pub const FRAME_EXEC_MAGIC: u32 = 0xA1B2_C3D4;
18
19/// Maximum number of timestamp pairs (begin + end per pass).
20const MAX_PASSES: u32 = 40;
21const MAX_QUERIES: u32 = MAX_PASSES * 2;
22/// `device.poll(Wait)` blocks inside the driver for up to 60s and can fail
23/// to return when a query resolve waits on work that is still queued.
24const GPU_WAIT: std::time::Duration = std::time::Duration::from_secs(5);
25
26/// Poll until the queue is idle, or fail after 5s. The `point` names the wait
27/// in the error. This does not call `PollType::Wait`.
28pub fn wait_for_gpu(device: &wgpu::Device, point: &str) -> Result<(), String> {
29    let deadline = std::time::Instant::now() + GPU_WAIT;
30    loop {
31        match device.poll(wgpu::PollType::Poll) {
32            Ok(wgpu::PollStatus::QueueEmpty | wgpu::PollStatus::WaitSucceeded) => return Ok(()),
33            Ok(wgpu::PollStatus::Poll) => {
34                if std::time::Instant::now() >= deadline {
35                    return Err(format!(
36                        "GPU wait timed out after 5s at {point}. The submission did not finish."
37                    ));
38                }
39                std::thread::sleep(std::time::Duration::from_millis(1));
40            }
41            Err(err) => return Err(format!("device.poll failed at {point}: {err}")),
42        }
43    }
44}

46/// How a begin/end pair was read.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum TimestampKind {
49    /// `end > begin` and the span is a positive number of milliseconds.
50    Measured,
51    /// Both ticks were written and they are equal. The pass was shorter than
52    /// one timestamp tick (or the driver collapsed the pair). Not a failure.
53    Empty,
54    /// Both ticks are 0. The query was resolved but never written.
55    Unwritten,
56    /// `end < begin`. Not a wrap of a full counter in one pass.
57    Inverted,
58    /// The device period is not positive, or a positive tick span underflows.
59    BadPeriod,
60}

62/// Per-pass GPU timing result.
63#[derive(Debug, Clone)]
64pub struct PassTiming {
65    pub name: String,
66    /// Milliseconds for [`TimestampKind::Measured`]. Zero for every other kind.
67    pub duration_ms: f32,
68    pub begin_tick: u64,
69    pub end_tick: u64,
70    pub kind: TimestampKind,
71}

73/// Pass sum plus the optional whole-frame bracket.
74#[derive(Debug, Clone)]
75pub struct GpuStampSummary {
76    /// Sum of measured passes other than the `frame` bracket.
77    pub pass_sum_ms: f32,
78    /// Encoder begin/end around `draw_into`, when that pair was measured.
79    pub frame_ms: Option<f32>,
80    pub empty: Vec<String>,
81    pub report: String,
82}
83
84fn format_stamp_report(period_ns: f32, results: &[PassTiming]) -> String {
85    let mut lines = vec![format!("timestamp_period_ns={period_ns}")];
86    for timing in results {
87        lines.push(format!(
88            "pass={} kind={:?} begin={} end={} duration_ms={:.6}",
89            timing.name, timing.kind, timing.begin_tick, timing.end_tick, timing.duration_ms
90        ));
91    }
92    if results.is_empty() {
93        lines.push("pass=(none)".to_string());
94    }
95    lines.join("\n")
96}

98fn summarize_stamps(period_ns: f32, results: &[PassTiming]) -> Result<GpuStampSummary, String> {
99    let report = format_stamp_report(period_ns, results);
100    if results.is_empty() {
101        return Err(format!("no GPU timestamp samples\n{report}"));
102    }
103    let mut pass_sum_ms = 0.0f32;
104    let mut frame_ms = None;
105    let mut empty = Vec::new();
106    let mut bad = Vec::new();
107    for timing in results {
108        match timing.kind {
109            TimestampKind::Measured if timing.name == "frame" => {
110                frame_ms = Some(timing.duration_ms);
111            }
112            TimestampKind::Measured => {
113                pass_sum_ms += timing.duration_ms;
114            }
115            TimestampKind::Empty => empty.push(timing.name.clone()),
116            TimestampKind::Unwritten | TimestampKind::Inverted | TimestampKind::BadPeriod => {
117                bad.push(format!("{} {:?}", timing.name, timing.kind));
118            }
119        }
120    }
121    if !bad.is_empty() {
122        return Err(format!(
123            "GPU timestamp read failed ({})\n{report}",
124            bad.join(", ")
125        ));
126    }
127    if !strictly_positive(pass_sum_ms) && frame_ms.is_none() {
128        return Err(format!(
129            "no pass recorded a positive GPU duration\n{report}"
130        ));
131    }
132    Ok(GpuStampSummary {
133        pass_sum_ms,
134        frame_ms,
135        empty,
136        report,
137    })
138}

140fn strictly_positive(value: f32) -> bool {
141    matches!(value.partial_cmp(&0.0), Some(std::cmp::Ordering::Greater))
142}

144/// Classify a raw pair. Equal non-zero ticks are [`TimestampKind::Empty`].
145pub fn classify_timestamp(begin: u64, end: u64, period_ns: f32) -> (TimestampKind, f32) {
146    match timestamp_duration_ms(begin, end, period_ns) {
147        Ok(ms) => (TimestampKind::Measured, ms),
148        Err("timestamp queries were not written") => (TimestampKind::Unwritten, 0.0),
149        Err("timestamp end is before begin") => (TimestampKind::Inverted, 0.0),
150        Err("GPU duration is zero or negative") if begin == end => (TimestampKind::Empty, 0.0),
151        Err(_) => (TimestampKind::BadPeriod, 0.0),
152    }
153}

155/// Convert a begin/end timestamp pair into milliseconds.
156///
157/// A zero period, an unwritten pair (both ticks 0), a NaN, or a non-positive
158/// span is not a measurement. Equal non-zero ticks are a zero span, not an
159/// unread query.
160pub fn timestamp_duration_ms(begin: u64, end: u64, period_ns: f32) -> Result<f32, &'static str> {
161    if !strictly_positive(period_ns) {
162        return Err("timestamp period is not positive");
163    }
164    if begin == 0 && end == 0 {
165        return Err("timestamp queries were not written");
166    }
167    if end < begin {
168        return Err("timestamp end is before begin");
169    }
170    let ms = ((end - begin) as f64) * f64::from(period_ns) / 1_000_000.0;
171    if !strictly_positive(ms as f32) {
172        return Err("GPU duration is zero or negative");
173    }
174    Ok(ms as f32)
175}

208fn resource_token<T: std::hash::Hash>(kind: u8, value: &T) -> u64 {
209    use std::collections::hash_map::DefaultHasher;
210    use std::hash::{Hash, Hasher};
211    let mut hasher = DefaultHasher::new();
212    kind.hash(&mut hasher);
213    value.hash(&mut hasher);
214    hasher.finish()
215}
216
217fn query_set_token(set: &wgpu::QuerySet) -> u64 {
218    resource_token(1, set)
219}

353    /// Replace the query set and both timestamp buffers.
354    ///
355    /// The self-test reads a set that has never been written. The frame path
356    /// reuses one set, and the query count changes after the first frame
357    /// (cluster and the far cascade drop out). A fresh set, resolve buffer,
358    /// and staging buffer make one sample match that self-test.
359    pub fn recreate(&mut self, device: &wgpu::Device) -> Result<(), &'static str> {
360        if self.readback_pending {
361            return Err("cannot recreate timestamp targets while the staging buffer is mapped");
362        }
363        self.query_set = device.create_query_set(&wgpu::QuerySetDescriptor {
364            label: Some("gpu_profiler_queries"),
365            ty: wgpu::QueryType::Timestamp,
366            count: MAX_QUERIES,
367        });
368        let buf_size = (MAX_QUERIES as u64) * std::mem::size_of::<u64>() as u64;
369        self.resolve_buf = device.create_buffer(&wgpu::BufferDescriptor {
370            label: Some("gpu_profiler_resolve"),
371            size: buf_size,
372            usage: wgpu::BufferUsages::QUERY_RESOLVE | wgpu::BufferUsages::COPY_SRC,
373            mapped_at_creation: false,
374        });
375        self.readback_buf = device.create_buffer(&wgpu::BufferDescriptor {
376            label: Some("gpu_profiler_readback"),
377            size: buf_size,
378            usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
379            mapped_at_creation: false,
380        });
381        self.next_query = 0;
382        self.pass_queries.clear();
383        self.stamp_len = 0;
384        self.latest_results.clear();
385        self.map_status = "none";
386        self.raw_len = 0;
387        self.parsed_queries = 0;
388        self.copy_size = 0;
389        Ok(())
390    }

399    fn note_stamp(
400        &mut self,
401        kind: StampKind,
402        index: u32,
403        label: &'static str,
404        submit: u64,
405        encoder: u64,
406    ) {
407        let len = self.stamp_len as usize;
408        if len >= STAMP_CAP {
409            return;
410        }
411        self.stamps[len] = StampRecord {
412            kind,
413            query_set: query_set_token(&self.query_set),
414            index,
415            label,
416            submit,
417            encoder,
418        };
419        self.stamp_len += 1;
420    }

422    /// `write_timestamp` on `encoder` for the begin index of `name`.
423    ///
424    /// This is the self-test's write. Pass descriptors are not used, so the
425    /// write cannot be dropped while the pass is built, and it lands in this
426    /// encoder rather than a pass-local command buffer.
427    pub fn begin_stamp(
428        &mut self,
429        encoder: &mut wgpu::CommandEncoder,
430        name: &'static str,
431    ) -> Option<OpenStamp> {
432        let (begin, end) = self.allocate_pass(name)?;
433        self.note_stamp(StampKind::Write, begin, name, 0, encoder_id(encoder));
434        let query_set = self.query_set.clone();
435        encoder.write_timestamp(&query_set, begin);
436        Some(OpenStamp { name, end })
437    }

439    /// Matching end `write_timestamp` for [`Self::begin_stamp`], on the same encoder.
440    pub fn end_stamp(&mut self, encoder: &mut wgpu::CommandEncoder, stamp: OpenStamp) {
441        self.note_stamp(
442            StampKind::Write,
443            stamp.end,
444            stamp.name,
445            0,
446            encoder_id(encoder),
447        );
448        let query_set = self.query_set.clone();
449        encoder.write_timestamp(&query_set, stamp.end);
450    }

466    /// One line per write, plus the resolve and copy submits.
467    pub fn submission_report(&self) -> String {
468        let mut lines = Vec::with_capacity(self.stamp_len as usize + 1);
469        lines.push(format!("query_set={}", query_set_token(&self.query_set)));
470        for stamp in self.stamp_trace() {
471            lines.push(format!(
472                "submit={} encoder={} kind={:?} label={} index={} query_set={}",
473                stamp.submit, stamp.encoder, stamp.kind, stamp.label, stamp.index, stamp.query_set
474            ));
475        }
476        let mut encoder_ids = [0u64; STAMP_CAP];
477        let mut encoder_count = 0usize;
478        for stamp in self.stamp_trace() {
479            if stamp.kind == StampKind::Copy || stamp.encoder == 0 {
480                continue;
481            }
482            if !encoder_ids[..encoder_count].contains(&stamp.encoder) {
483                encoder_ids[encoder_count] = stamp.encoder;
484                encoder_count += 1;
485            }
486        }
487        lines.push(format!("encoder_count={encoder_count}"));
488        if self.stamp_len == 0 {
489            lines.push("submit=(none)".to_string());
490        }
491        lines.push(format!(
492            "resolve_buffer={} resolve_offset={}",
493            self.resolve_buffer, self.resolve_offset
494        ));
495        lines.push(format!(
496            "copy src={} src_offset={} dst={} dst_offset={} size={}",
497            self.copy_src,
498            self.copy_src_offset,
499            self.copy_dst,
500            self.copy_dst_offset,
501            self.copy_size
502        ));
503        lines.push(format!(
504            "mapped_buffer={} map={} parsed_queries={}",
505            self.mapped_buffer, self.map_status, self.parsed_queries
506        ));
507        let shown = self.raw_len as usize;
508        let raw = self.raw_ticks[..shown]
509            .iter()
510            .map(|tick| tick.to_string())
511            .collect::<Vec<_>>()
512            .join(",");
513        lines.push(format!("raw_u64={raw}"));
514        lines.join("\n")
515    }

525    /// Allocate a begin/end query index pair for a named pass.
526    /// Returns `(begin_index, end_index)` or `None` if slots are exhausted.
527    /// The returned indices can be used with `query_set()` to construct
528    /// `RenderPassTimestampWrites` or `ComputePassTimestampWrites` inline.
529    pub fn allocate_pass(&mut self, name: &'static str) -> Option<(u32, u32)> {
530        if self.next_query + 2 > MAX_QUERIES {
531            return None;
532        }
533        let begin = self.next_query;
534        let end = self.next_query + 1;
535        self.next_query += 2;
536        self.pass_queries.push((name, begin, end));
537        Some((begin, end))
538    }

584    /// This frame wrote queries and the staging buffer is free to receive them.
585    pub fn staging_copy_pending(&self) -> bool {
586        self.next_query > 0 && !self.readback_pending
587    }

589    /// Mark the frame closed. [`Self::resolve_queries`] records the resolve
590    /// on the write encoder before [`Self::submit_frame`] submits that buffer.
591    pub fn end_frame(&self, _encoder: &mut wgpu::CommandEncoder) {
592        if self.readback_pending && self.next_query > 0 {
593            log::debug!(
594                "gpu profiler skipped timestamp resolve; gpu_profiler_readback is still mapped"
595            );
596        }
597    }

599    /// Resolve written queries into the GPU buffer on `encoder`.
600    ///
601    /// `encoder` is the `CommandEncoder` that recorded this frame's
602    /// `write_timestamp` calls. wgpu tracks query resets per encoder. A
603    /// resolve on a different encoder reads queries that encoder never wrote,
604    /// and NVIDIA returns 0 for them. The staging copy stays a later submission
605    /// (wgpu#6406).
606    pub fn resolve_queries(&mut self, encoder: &mut wgpu::CommandEncoder) {
607        if !self.staging_copy_pending() {
608            return;
609        }
610        if self
611            .stamp_trace()
612            .iter()
613            .any(|stamp| stamp.kind == StampKind::Resolve)
614        {
615            return;
616        }
617        self.resolve_buffer = resource_token(2, &self.resolve_buf);
618        self.resolve_offset = 0;
619        let index = self.next_query;
620        let recorded_on = encoder_id(encoder);
621        self.note_stamp(StampKind::Resolve, index, "resolve", 0, recorded_on);
622        encoder.resolve_query_set(&self.query_set, 0..index, &self.resolve_buf, 0);
623    }

625    /// Every `write_timestamp` and the one `resolve_query_set` share an encoder.
626    ///
627    /// A matching submit id is not enough: `queue.submit` can take two
628    /// command buffers, and that id does not say which encoder recorded the
629    /// resolve.
630    fn assert_shared_write_encoder(&self) -> Result<(), String> {
631        let mut resolve_encoder = 0u64;
632        let mut resolve_count = 0u32;
633        for stamp in self.stamp_trace() {
634            if stamp.kind == StampKind::Resolve {
635                resolve_encoder = stamp.encoder;
636                resolve_count += 1;
637            }
638        }
639        if resolve_count != 1 || resolve_encoder == 0 {
640            return Err(format!(
641                "expected one resolve_query_set on the write encoder, found {resolve_count}\n{}",
642                self.submission_report()
643            ));
644        }
645        for stamp in self.stamp_trace() {
646            if stamp.kind != StampKind::Write {
647                continue;
648            }
649            if stamp.encoder != resolve_encoder {
650                return Err(format!(
651                    "timestamp write {} index {} encoder {} != resolve encoder {}\n{}",
652                    stamp.label,
653                    stamp.index,
654                    stamp.encoder,
655                    resolve_encoder,
656                    self.submission_report()
657                ));
658            }
659        }
660        Ok(())
661    }

663    /// Copy the resolved timestamps into the CPU staging buffer.
664    ///
665    /// Submit this encoder only after the write-and-resolve submission is idle
666    /// (wgpu#6406). It is not the encoder that wrote the timestamps.
667    pub fn copy_resolved(&mut self, encoder: &mut wgpu::CommandEncoder) {
668        if !self.staging_copy_pending() {
669            return;
670        }
671        let byte_count = (self.next_query as u64) * std::mem::size_of::<u64>() as u64;
672        self.copy_src = resource_token(2, &self.resolve_buf);
673        self.copy_dst = resource_token(3, &self.readback_buf);
674        self.copy_src_offset = 0;
675        self.copy_dst_offset = 0;
676        self.copy_size = byte_count;
677        encoder.copy_buffer_to_buffer(&self.resolve_buf, 0, &self.readback_buf, 0, byte_count);
678    }

680    /// Submit the command buffer that already contains the writes and the resolve.
681    ///
682    /// [`Self::resolve_queries`] must have run on that same `CommandEncoder`.
683    /// Opening a second encoder for the resolve submits two Vulkan command
684    /// buffers. wgpu resets queries per command buffer, so the resolve reads
685    /// queries that buffer never wrote, and NVIDIA returns 0. The submit id
686    /// is logical and does not show that split. The copy is a later submission
687    /// (wgpu#6406). Resolving only after a wait of the writes deadlocks NVIDIA:
688    /// `VK_QUERY_RESULT_WAIT_BIT` runs while those writes are still queued.
689    /// Every wait is a non-blocking poll with a 5s limit.
690    pub fn submit_frame(
691        profiler: Option<&mut Self>,
692        queue: &wgpu::Queue,
693        device: &wgpu::Device,
694        frame: wgpu::CommandBuffer,
695    ) -> Result<(), String> {
696        let Some(profiler) = profiler else {
697            queue.submit(std::iter::once(frame));
698            return wait_for_gpu(device, "frame submit");
699        };
700        if !profiler.staging_copy_pending() {
701            queue.submit(std::iter::once(frame));
702            return wait_for_gpu(device, "frame submit");
703        }
704        profiler.assert_shared_write_encoder()?;
705        profiler.submit_serial += 1;
706        let write_submit = profiler.submit_serial;
707        profiler.assign_open_stamps(write_submit);
708        queue.submit(std::iter::once(frame));
709        let write_started = std::time::Instant::now();
710        wait_for_gpu(device, "timestamp write+resolve")?;
711        profiler.write_wait_ms = write_started.elapsed().as_secs_f32() * 1_000.0;
712        let mut copy = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
713            label: Some("gpu profiler timestamp copy"),
714        });
715        let copy_encoder = encoder_id(&copy);
716        profiler.copy_resolved(&mut copy);
717        profiler.submit_serial += 1;
718        let copy_submit = profiler.submit_serial;
719        profiler.note_stamp(
720            StampKind::Copy,
721            profiler.next_query,
722            "copy",
723            copy_submit,
724            copy_encoder,
725        );
726        queue.submit(std::iter::once(copy.finish()));
727        let copy_started = std::time::Instant::now();
728        let copied = wait_for_gpu(device, "timestamp staging copy");
729        profiler.copy_wait_ms = copy_started.elapsed().as_secs_f32() * 1_000.0;
730        copied
731    }

733    /// Initiate an async readback of the timestamp buffer.
734    /// Call after `queue.submit()`.
735    pub fn request_readback(&mut self) {
736        if self.next_query == 0 || self.readback_pending {
737            return;
738        }
739        let slice = self.readback_buf.slice(..);
740        self.mapped_buffer = resource_token(3, &self.readback_buf);
741        self.map_status = "pending";
742        self.readback_pending = true;
743        self.last_error = None;
744        self.map_ready.store(false, Ordering::Release);
745        self.map_ok.store(false, Ordering::Release);
746        let ready = self.map_ready.clone();
747        let ok = self.map_ok.clone();
748        slice.map_async(wgpu::MapMode::Read, move |result| {
749            ok.store(result.is_ok(), Ordering::Release);
750            ready.store(true, Ordering::Release);
751        });
752    }

754    /// Wait until the mapped timestamp buffer is ready, then decode it.
755    ///
756    /// Uses [`wait_for_gpu`]'s 5s poll, not `PollType::Wait`. The map callback
757    /// only stores an atomic, so this thread can poll it.
758    pub fn block_on_readback(&mut self, device: &wgpu::Device) -> Result<(), String> {
759        if !self.readback_pending {
760            return Err("timestamp readback was not requested".to_string());
761        }
762        let deadline = std::time::Instant::now() + GPU_WAIT;
763        loop {
764            if let Err(err) = device.poll(wgpu::PollType::Poll) {
765                return Err(format!(
766                    "device.poll failed at timestamp map readback: {err}"
767                ));
768            }
769            if self.poll_readback(device) {
770                return Ok(());
771            }
772            if !self.readback_pending {
773                return Err(self
774                    .last_error
775                    .unwrap_or("timestamp readback did not complete")
776                    .to_string());
777            }
778            if std::time::Instant::now() >= deadline {
779                return Err(
780                    "GPU wait timed out after 5s at timestamp map readback. The submission did not finish."
781                        .to_string(),
782                );
783            }
784            std::thread::sleep(std::time::Duration::from_millis(1));
785        }
786    }

788    /// Poll for completed readback and update `latest_results`.
789    /// Returns `true` if new data is available.
790    pub fn poll_readback(&mut self, device: &wgpu::Device) -> bool {
791        if !self.readback_pending {
792            return false;
793        }
794        // Non-blocking poll — drives the map callback.
795        if device.poll(wgpu::PollType::Poll).is_err() {
796            log::error!("gpu profiler device poll failed; not reading a mapped buffer");
797            return false;
798        }
799
800        if !self.map_ready.load(Ordering::Acquire) {
801            // Still mapped. end_frame will not copy into this buffer.
802            return false;
803        }
804        if !self.map_ok.load(Ordering::Acquire) {
805            self.readback_pending = false;
806            self.map_status = "err";
807            self.last_error = Some("timestamp buffer map failed");
808            self.latest_results.clear();
809            return false;
810        }
811
812        let slice = self.readback_buf.slice(..);
813        let data = slice.get_mapped_range();
814        let raw_bytes = (16 * std::mem::size_of::<u64>()).min(data.len());
815        let raw: &[u64] = bytemuck::cast_slice(&data[..raw_bytes]);
816        self.raw_ticks = [0; 16];
817        self.raw_len = raw.len() as u8;
818        self.raw_ticks[..raw.len()].copy_from_slice(raw);
819        self.parsed_queries = self.next_query;
820        self.map_status = "ok";
821        let timestamps: &[u64] =
822            bytemuck::cast_slice(&data[..self.next_query as usize * std::mem::size_of::<u64>()]);
823
824        self.latest_results.clear();
825        for (name, begin_idx, end_idx) in &self.pass_queries {
826            let t0 = timestamps[*begin_idx as usize];
827            let t1 = timestamps[*end_idx as usize];
828            let (kind, duration_ms) = classify_timestamp(t0, t1, self.timestamp_period);
829            self.latest_results.push(PassTiming {
830                name: (*name).to_string(),
831                duration_ms,
832                begin_tick: t0,
833                end_tick: t1,
834                kind,
835            });
836        }
837
838        drop(data);
839        self.readback_buf.unmap();
840        self.readback_pending = false;
841        self.map_ready.store(false, Ordering::Release);
842        true
843    }

850    /// Write two encoder timestamps around a 1×1 clear, resolve them on that
851    /// same encoder, then use [`Self::submit_frame`]. The copy is the next
852    /// submission. `Ok` includes `encoder_count`.
853    ///
854    /// `Ok` is the stamp report. A `0, 0` pair is a failure. Equal non-zero
855    /// ticks are [`TimestampKind::Empty`]: a trivial pass can be shorter than
856    /// one timestamp tick, and that is not a failure.
857    pub fn prove_readback(device: &wgpu::Device, queue: &wgpu::Queue) -> Result<String, String> {
858        let features = device.features();
859        if !features.contains(wgpu::Features::TIMESTAMP_QUERY) {
860            return Err("timestamp self-test: device is missing TIMESTAMP_QUERY".to_string());
861        }
862        if !features.contains(wgpu::Features::TIMESTAMP_QUERY_INSIDE_ENCODERS) {
863            return Err(
864                "timestamp self-test: device is missing TIMESTAMP_QUERY_INSIDE_ENCODERS, so write_timestamp cannot run"
865                    .to_string(),
866            );
867        }
868        let mut profiler = Self::new(device, queue);
869        let period = profiler.timestamp_period_ns();
870        if !strictly_positive(period) {
871            return Err(format!(
872                "timestamp self-test: timestamp_period_ns={period} is not positive"
873            ));
874        }
875        profiler.begin_frame();
876        let target = device.create_texture(&wgpu::TextureDescriptor {
877            label: Some("timestamp self-test target"),
878            size: wgpu::Extent3d {
879                width: 1,
880                height: 1,
881                depth_or_array_layers: 1,
882            },
883            mip_level_count: 1,
884            sample_count: 1,
885            dimension: wgpu::TextureDimension::D2,
886            format: wgpu::TextureFormat::Rgba8Unorm,
887            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
888            view_formats: &[],
889        });
890        let view = target.create_view(&wgpu::TextureViewDescriptor::default());
891
892        let mut writes = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
893            label: Some("timestamp self-test writes"),
894        });
895        let outer = profiler
896            .begin_stamp(&mut writes, "self_test")
897            .ok_or_else(|| {
898                "timestamp self-test: no query slot for the encoder bracket".to_string()
899            })?;
900        let clear = profiler
901            .begin_stamp(&mut writes, "self_test_clear")
902            .ok_or_else(|| "timestamp self-test: no query slot for the clear pass".to_string())?;
903        {
904            let _pass = writes.begin_render_pass(&wgpu::RenderPassDescriptor {
905                label: Some("timestamp self-test clear"),
906                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
907                    view: &view,
908                    resolve_target: None,
909                    ops: wgpu::Operations {
910                        load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
911                        store: wgpu::StoreOp::Store,
912                    },
913                })],
914                depth_stencil_attachment: None,
915                timestamp_writes: None,
916                occlusion_query_set: None,
917            });
918        }
919        profiler.end_stamp(&mut writes, clear);
920        profiler.end_stamp(&mut writes, outer);
921        profiler.resolve_queries(&mut writes);
922        profiler.end_frame(&mut writes);
923
924        let frame = writes.finish();
925        Self::submit_frame(Some(&mut profiler), queue, device, frame)
926            .map_err(|err| format!("timestamp self-test: {err}\ntimestamp_period_ns={period}"))?;
927        profiler.request_readback();
928        if let Err(err) = profiler.block_on_readback(device) {
929            return Err(format!(
930                "timestamp self-test: {err}\n{}",
931                profiler.timestamp_report()
932            ));
933        }
934        let report = profiler.timestamp_report();
935        let unread: Vec<&str> = profiler
936            .results()
937            .iter()
938            .filter(|timing| timing.begin_tick == 0 && timing.end_tick == 0)
939            .map(|timing| timing.name.as_str())
940            .collect();
941        if !unread.is_empty() {
942            return Err(format!(
943                "timestamp self-test read zeros ({}); resolve_query_set must be on the write encoder and the copy is a later submission\n{report}\n{}",
944                unread.join(", "),
945                profiler.submission_report()
946            ));
947        }
948        if profiler.results().is_empty() {
949            return Err(format!("timestamp self-test read no slots\n{report}"));
950        }
951        Ok(format!("{report}\n{}", profiler.submission_report()))
952    }
