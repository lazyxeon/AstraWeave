24/// Planner mode selection for AI entities.
25#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
26#[non_exhaustive]
27pub enum PlannerMode {
28    /// Use rule-based orchestrator (smoke + advance logic).
29    Rule,
30    /// Use behavior tree planner (requires `ai-bt` feature).
31    BehaviorTree,
32    /// Use GOAP planner (requires `ai-goap` feature).
33    GOAP,
34    /// Hierarchical planner with utility-scored methods.
35    Htn,
36}
37
38impl std::fmt::Display for PlannerMode {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        match self {
41            PlannerMode::Rule => write!(f, "Rule"),
42            PlannerMode::BehaviorTree => write!(f, "BehaviorTree"),
43            PlannerMode::GOAP => write!(f, "GOAP"),
44            PlannerMode::Htn => write!(f, "Htn"),
45        }
46    }
47}
48
49impl PlannerMode {
50    /// Check if this mode requires the `ai-bt` feature.
51    #[must_use]
52    pub fn requires_bt_feature(&self) -> bool {
53        matches!(self, PlannerMode::BehaviorTree)
54    }
55
56    /// Check if this mode requires the `ai-goap` feature.
57    #[must_use]
58    pub fn requires_goap_feature(&self) -> bool {
59        matches!(self, PlannerMode::GOAP)
60    }
61
62    /// Check if this mode is always available (no feature flag required).
63    #[must_use]
64    pub fn is_always_available(&self) -> bool {
65        matches!(self, PlannerMode::Rule | PlannerMode::Htn)
66    }
67
68    /// Get the feature flag name required for this mode, if any.
69    #[must_use]
70    pub fn required_feature(&self) -> Option<&'static str> {
71        match self {
72            PlannerMode::Rule | PlannerMode::Htn => None,
73            PlannerMode::BehaviorTree => Some("ai-bt"),
74            PlannerMode::GOAP => Some("ai-goap"),
75        }
76    }
77
78    /// Get all available planner modes.
79    #[must_use]
80    pub fn all() -> &'static [PlannerMode] {
81        &[
82            PlannerMode::Rule,
83            PlannerMode::BehaviorTree,
84            PlannerMode::GOAP,
85            PlannerMode::Htn,
86        ]
87    }
88}

90/// Component for AI entities to control which planner to use.
91///
92/// # Example
93/// ```
94/// use astraweave_ai::core_loop::{CAiController, PlannerMode};
95/// use astraweave_ai::htn::GatherGoal;
96///
97/// let controller = CAiController {
98///     mode: PlannerMode::GOAP,
99///     policy: Some("gather_craft_policy".to_string()),
100///     gather: GatherGoal::KeepWood,
101/// };
102/// ```
103#[derive(Clone, Debug, PartialEq, Eq)]
104pub struct CAiController {
105    /// Which planner to use for this entity.
106    pub mode: PlannerMode,
107    /// Optional behavior-tree policy name. GOAP dispatch does not read it.
108    pub policy: Option<String>,
109    /// Explicit gather method. Ignored by rule and behavior-tree modes.
110    pub gather: GatherGoal,
111}
112
113impl std::fmt::Display for CAiController {
114    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
115        match &self.policy {
116            Some(policy) => write!(f, "CAiController({}, policy={})", self.mode, policy),
117            None => write!(f, "CAiController({})", self.mode),
118        }
119    }
120}
121
122impl Default for CAiController {
123    fn default() -> Self {
124        Self {
125            mode: PlannerMode::Rule,
126            policy: None,
127            gather: GatherGoal::CookFood,
128        }
129    }
130}

207/// Dispatch planning to the appropriate planner based on controller mode.
208///
209/// # Arguments
210/// - `controller`: The AI controller specifying which planner to use
211/// - `snapshot`: World state snapshot for the entity
212///
213/// # Returns
214/// - `Ok(PlanIntent)` with the planned action sequence
215/// - `Err` if planning fails
216///
217/// # Example
218/// ```
219/// use astraweave_ai::core_loop::{CAiController, PlannerMode, dispatch_planner};
220/// use astraweave_core::{WorldSnapshot, PlayerState, CompanionState, EnemyState, IVec2, WeaveHazard, MAX_WEAVE_HAZARDS};
221/// use std::collections::BTreeMap;
222///
223/// let controller = CAiController {
224///     mode: PlannerMode::Rule,
225///     policy: None,
226///     gather: astraweave_ai::htn::GatherGoal::CookFood,
227/// };
228///
229/// // Create snapshot with an enemy so the planner returns non-empty steps
230/// let snapshot = WorldSnapshot {
231///     t: 0.0,
232///     me: CompanionState {
233///         ammo: 10,
234///         cooldowns: BTreeMap::new(),
235///         morale: 1.0,
236///         pos: IVec2 { x: 5, y: 5 },
237///     },
238///     player: PlayerState {
239///         hp: 100,
240///         pos: IVec2 { x: 5, y: 5 },
241///         stance: "stand".into(),
242///         orders: vec![],
243///     },
244///     enemies: vec![EnemyState {
245///         id: 1,
246///         pos: IVec2 { x: 10, y: 10 },
247///         hp: 100,
248///         cover: "none".into(),
249///         last_seen: 0.0,
250///     }],
251///     pois: vec![],
252///     obstacles: vec![],
253///     objective: None,
254///     hazards: [WeaveHazard::EMPTY; MAX_WEAVE_HAZARDS],
255///     hazard_count: 0,
256/// };
257///
258/// let plan = dispatch_planner(&controller, &snapshot).unwrap();
259/// assert!(!plan.steps.is_empty());
260/// ```
261pub fn dispatch_planner(
262    controller: &CAiController,
263    snapshot: &WorldSnapshot,
264) -> Result<PlanIntent> {
265    #[cfg(feature = "profiling")]
266    span!("AI::dispatch_planner");
267
268    match controller.mode {
269        PlannerMode::Rule => {
270            let orch = RuleOrchestrator;
271            Ok(orch.propose_plan(snapshot))
272        }
273        PlannerMode::BehaviorTree => {
274            #[cfg(feature = "ai-bt")]
275            {
276                dispatch_bt(controller, snapshot)
277            }
278            #[cfg(not(feature = "ai-bt"))]
279            {
280                anyhow::bail!("BehaviorTree mode requires 'ai-bt' feature")
281            }
282        }
283        PlannerMode::GOAP => {
284            #[cfg(feature = "ai-goap")]
285            {
286                dispatch_goap(controller, snapshot)
287            }
288            #[cfg(not(feature = "ai-goap"))]
289            {
290                anyhow::bail!("GOAP mode requires 'ai-goap' feature")
291            }
292        }
293        PlannerMode::Htn => Ok(crate::htn::HtnOrchestrator.propose_plan(snapshot)),
294    }
295}

297/// Typed failure from behavior-tree dispatch.
298///
299/// Returned when `ai-bt` is enabled and the controller does not name a graph
300/// this dispatcher can tick. The default policy `"default"` ticks the built-in
301/// companion tree inside `dispatch_planner`.
302#[derive(Debug, Clone, PartialEq, Eq)]
303pub struct BehaviorDispatchError {
304    message: String,
305}

310        Self {
311            message: message.into(),
312        }
313    }
314}

318        f.write_str(&self.message)
319    }

324/// Dispatch to the behavior-tree planner.
325///
326/// Policy `"default"` ticks a selector: advance toward `enemies.first()` when
327/// an enemy is present, otherwise hold. Any other policy is a typed error.
328/// The graph walk is the behavior runtime; it does not call the model.
329#[cfg(feature = "ai-bt")]
330fn dispatch_bt(controller: &CAiController, snapshot: &WorldSnapshot) -> Result<PlanIntent> {
331    match controller.policy.as_deref() {
332        Some("default") => Ok(tick_default_companion_tree(snapshot)),
333        Some(policy) => Err(BehaviorDispatchError::new(format!(
334            "no behavior graph is bound for policy '{policy}'"
335        ))
336        .into()),
337        None => {
338            Err(BehaviorDispatchError::new("no behavior graph is bound to this controller").into())
339        }
340    }
341}
342
343#[cfg(feature = "ai-bt")]
344fn tick_default_companion_tree(snapshot: &WorldSnapshot) -> PlanIntent {
345    use std::sync::{Arc, Mutex};
346
347    let steps = Arc::new(Mutex::new(Vec::<ActionStep>::new()));
348    let enemy = snapshot.enemies.first().cloned();
349    let me = snapshot.me.pos;
350    let enemy_present = enemy.is_some();
351
352    let mut context = BehaviorContext::new();
353    context.register_condition("enemy_present", move || enemy_present);
354    let advance_steps = Arc::clone(&steps);
355    context.register_action("advance", move || {
356        if let Some(enemy) = &enemy {
357            let x = me.x.saturating_add((enemy.pos.x - me.x).signum());
358            let y = me.y.saturating_add((enemy.pos.y - me.y).signum());
359            if let Ok(mut recorded) = advance_steps.lock() {
360                recorded.push(ActionStep::MoveTo { x, y, speed: None });
361            }
362        }
363        BehaviorStatus::Success
364    });
365    context.register_action("hold", || BehaviorStatus::Success);
366
367    let root = BehaviorNode::Selector(vec![
368        BehaviorNode::Sequence(vec![
369            BehaviorNode::Condition("enemy_present".into()),
370            BehaviorNode::Action("advance".into()),
371        ]),
372        BehaviorNode::Action("hold".into()),
373    ]);
374    let _status = BehaviorGraph::new(root).tick(&context);
375    let recorded = steps.lock().map(|guard| guard.clone()).unwrap_or_default();
376    PlanIntent {
377        plan_id: format!("bt-{}", (snapshot.t * 1000.0) as i64),
378        steps: recorded,
379    }
380}

382/// Gather methods live in the HTN module. The feature gate stays so existing
383/// controllers still select this mode by name.
384#[cfg(feature = "ai-goap")]
385fn dispatch_goap(controller: &CAiController, _snapshot: &WorldSnapshot) -> Result<PlanIntent> {
386    #[cfg(feature = "profiling")]
387    span!("AI::dispatch_goap");
388    Ok(crate::htn::gather_method(controller.gather))
389}
