25/// Planner mode selection for AI entities.
26#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
27#[non_exhaustive]
28pub enum PlannerMode {
29    /// Use rule-based orchestrator (smoke + advance logic).
30    Rule,
31    /// Use behavior tree planner (requires `ai-bt` feature).
32    BehaviorTree,
33    /// Use GOAP planner (requires `ai-goap` feature).
34    GOAP,
35    /// Hierarchical planner with utility-scored methods.
36    Htn,
37}
38
39impl std::fmt::Display for PlannerMode {
40    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
41        match self {
42            PlannerMode::Rule => write!(f, "Rule"),
43            PlannerMode::BehaviorTree => write!(f, "BehaviorTree"),
44            PlannerMode::GOAP => write!(f, "GOAP"),
45            PlannerMode::Htn => write!(f, "Htn"),
46        }
47    }
48}
49
50impl PlannerMode {
51    /// Check if this mode requires the `ai-bt` feature.
52    #[must_use]
53    pub fn requires_bt_feature(&self) -> bool {
54        matches!(self, PlannerMode::BehaviorTree)
55    }
56
57    /// Check if this mode requires the `ai-goap` feature.
58    #[must_use]
59    pub fn requires_goap_feature(&self) -> bool {
60        matches!(self, PlannerMode::GOAP)
61    }
62
63    /// Check if this mode is always available (no feature flag required).
64    #[must_use]
65    pub fn is_always_available(&self) -> bool {
66        matches!(self, PlannerMode::Rule | PlannerMode::Htn)
67    }
68
69    /// Get the feature flag name required for this mode, if any.
70    #[must_use]
71    pub fn required_feature(&self) -> Option<&'static str> {
72        match self {
73            PlannerMode::Rule | PlannerMode::Htn => None,
74            PlannerMode::BehaviorTree => Some("ai-bt"),
75            PlannerMode::GOAP => Some("ai-goap"),
76        }
77    }
78
79    /// Get all available planner modes.
80    #[must_use]
81    pub fn all() -> &'static [PlannerMode] {
82        &[
83            PlannerMode::Rule,
84            PlannerMode::BehaviorTree,
85            PlannerMode::GOAP,
86            PlannerMode::Htn,
87        ]
88    }
89}

91/// Component for AI entities to control which planner to use.
92///
93/// # Example
94/// ```
95/// use astraweave_ai::core_loop::{CAiController, PlannerMode};
96///
97/// let controller = CAiController {
98///     mode: PlannerMode::GOAP,
99///     policy: Some("gather_craft_policy".to_string()),
100/// };
101/// ```
102#[derive(Clone, Debug, PartialEq, Eq)]
103pub struct CAiController {
104    /// Which planner to use for this entity.
105    pub mode: PlannerMode,
106    /// Optional policy identifier for BT/GOAP configuration.
107    /// Can be used to load specific behavior trees or GOAP action sets.
108    pub policy: Option<String>,
109}
110
111impl std::fmt::Display for CAiController {
112    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
113        match &self.policy {
114            Some(policy) => write!(f, "CAiController({}, policy={})", self.mode, policy),
115            None => write!(f, "CAiController({})", self.mode),
116        }
117    }
118}
119
120impl Default for CAiController {
121    fn default() -> Self {
122        Self {
123            mode: PlannerMode::Rule,
124            policy: None,
125        }
126    }
127}

167    }

188        self.mode.required_feature().is_some()

192/// Dispatch planning to the appropriate planner based on controller mode.
193///
194/// # Arguments
195/// - `controller`: The AI controller specifying which planner to use
196/// - `snapshot`: World state snapshot for the entity
197///
198/// # Returns
199/// - `Ok(PlanIntent)` with the planned action sequence
200/// - `Err` if planning fails
201///
202/// # Example
203/// ```
204/// use astraweave_ai::core_loop::{CAiController, PlannerMode, dispatch_planner};
205/// use astraweave_core::{WorldSnapshot, PlayerState, CompanionState, EnemyState, IVec2};
206/// use std::collections::BTreeMap;
207///
208/// let controller = CAiController {
209///     mode: PlannerMode::Rule,
210///     policy: None,
211/// };
212///
213/// // Create snapshot with an enemy so the planner returns non-empty steps
214/// let snapshot = WorldSnapshot {
215///     t: 0.0,
216///     me: CompanionState {
217///         ammo: 10,
218///         cooldowns: BTreeMap::new(),
219///         morale: 1.0,
220///         pos: IVec2 { x: 5, y: 5 },
221///     },
222///     player: PlayerState {
223///         hp: 100,
224///         pos: IVec2 { x: 5, y: 5 },
225///         stance: "stand".into(),
226///         orders: vec![],
227///     },
228///     enemies: vec![EnemyState {
229///         id: 1,
230///         pos: IVec2 { x: 10, y: 10 },
231///         hp: 100,
232///         cover: "none".into(),
233///         last_seen: 0.0,
234///     }],
235///     pois: vec![],
236///     obstacles: vec![],
237///     objective: None,
238/// };
239///
240/// let plan = dispatch_planner(&controller, &snapshot).unwrap();
241/// assert!(!plan.steps.is_empty());
242/// ```
243pub fn dispatch_planner(
244    controller: &CAiController,
245    snapshot: &WorldSnapshot,
246) -> Result<PlanIntent> {
247    #[cfg(feature = "profiling")]
248    span!("AI::dispatch_planner");
249
250    match controller.mode {
251        PlannerMode::Rule => {
252            let orch = RuleOrchestrator;
253            Ok(orch.propose_plan(snapshot))
254        }
255        PlannerMode::BehaviorTree => {
256            #[cfg(feature = "ai-bt")]
257            {
258                dispatch_bt(controller, snapshot)
259            }
260            #[cfg(not(feature = "ai-bt"))]
261            {
262                anyhow::bail!("BehaviorTree mode requires 'ai-bt' feature")
263            }
264        }
265        PlannerMode::GOAP => {
266            #[cfg(feature = "ai-goap")]
267            {
268                dispatch_goap(controller, snapshot)
269            }
270            #[cfg(not(feature = "ai-goap"))]
271            {
272                anyhow::bail!("GOAP mode requires 'ai-goap' feature")
273            }
274        }
275        PlannerMode::Htn => Ok(crate::htn::HtnOrchestrator.propose_plan(snapshot)),
276    }
277}

306/// Dispatch to the behavior-tree planner.
307///
308/// Policy `"default"` ticks a selector: advance toward `enemies.first()` when
309/// an enemy is present, otherwise hold. Any other policy is a typed error.
310/// The graph walk is the behavior runtime; it does not call the model.
311#[cfg(feature = "ai-bt")]
312fn dispatch_bt(controller: &CAiController, snapshot: &WorldSnapshot) -> Result<PlanIntent> {
313    match controller.policy.as_deref() {
314        Some("default") => Ok(tick_default_companion_tree(snapshot)),
315        Some(policy) => Err(BehaviorDispatchError::new(format!(
316            "no behavior graph is bound for policy '{policy}'"
317        ))
318        .into()),
319        None => {
320            Err(BehaviorDispatchError::new("no behavior graph is bound to this controller").into())
321        }
322    }
323}
324
325#[cfg(feature = "ai-bt")]
326fn tick_default_companion_tree(snapshot: &WorldSnapshot) -> PlanIntent {
327    use std::sync::{Arc, Mutex};
328
329    let steps = Arc::new(Mutex::new(Vec::<ActionStep>::new()));
330    let enemy = snapshot.enemies.first().cloned();
331    let me = snapshot.me.pos;
332    let enemy_present = enemy.is_some();
333
334    let mut context = BehaviorContext::new();
335    context.register_condition("enemy_present", move || enemy_present);
336    let advance_steps = Arc::clone(&steps);
337    context.register_action("advance", move || {
338        if let Some(enemy) = &enemy {
339            let x = me.x.saturating_add((enemy.pos.x - me.x).signum());
340            let y = me.y.saturating_add((enemy.pos.y - me.y).signum());
341            if let Ok(mut recorded) = advance_steps.lock() {
342                recorded.push(ActionStep::MoveTo { x, y, speed: None });
343            }
344        }
345        BehaviorStatus::Success
346    });
347    context.register_action("hold", || BehaviorStatus::Success);
348
349    let root = BehaviorNode::Selector(vec![
350        BehaviorNode::Sequence(vec![
351            BehaviorNode::Condition("enemy_present".into()),
352            BehaviorNode::Action("advance".into()),
353        ]),
354        BehaviorNode::Action("hold".into()),
355    ]);
356    let _status = BehaviorGraph::new(root).tick(&context);
357    let recorded = steps.lock().map(|guard| guard.clone()).unwrap_or_default();
358    PlanIntent {
359        plan_id: format!("bt-{}", (snapshot.t * 1000.0) as i64),
360        steps: recorded,
361    }
362}

364/// Dispatch to GOAP planner (feature-gated).
365#[cfg(feature = "ai-goap")]
366fn dispatch_goap(controller: &CAiController, snapshot: &WorldSnapshot) -> Result<PlanIntent> {
367    #[cfg(feature = "profiling")]
368    span!("AI::dispatch_goap");
369
370    // Convert WorldSnapshot to GOAP WorldState
371    let world_state = snapshot_to_goap_state(snapshot)?;
372
373    // Define goal based on policy or default
374    let goal = if let Some(policy_name) = &controller.policy {
375        if policy_name.contains("gather") || policy_name.contains("craft") {
376            GoapGoal::new(
377                "gather_and_craft",
378                WorldState::from_facts(&[("has_wood", true), ("has_food", true)]),
379            )
380        } else {
381            GoapGoal::new("get_food", WorldState::from_facts(&[("has_food", true)]))
382        }
383    } else {
384        GoapGoal::new("get_food", WorldState::from_facts(&[("has_food", true)]))
385    };
386
387    // Define available actions
388    let actions = create_goap_actions();
389
390    // Run GOAP planner
391    let planner = GoapPlanner::new().with_max_iterations(100);
392    let plan = planner
393        .plan(&world_state, &goal, &actions)
394        .ok_or_else(|| anyhow::anyhow!("GOAP planning failed - no plan found"))?;
395
396    // Convert GOAP action sequence to ActionSteps
397    let steps = goap_actions_to_steps(&plan)?;
398
399    Ok(PlanIntent {
400        plan_id: format!("goap_{}", plan.len()),
401        steps,
402    })
403}

405/// Convert WorldSnapshot to GOAP WorldState.
406#[cfg(feature = "ai-goap")]
407fn snapshot_to_goap_state(snapshot: &WorldSnapshot) -> Result<WorldState> {
408    let mut state = WorldState::new();
409    state.set("has_wood", false);
410    state.set("has_food", false);
411    state.set("at_tree", false);
412    state.set("at_campfire", false);
413    state.set("hungry", snapshot.player.hp < 50);
414    Ok(state)
415}

437            .with_effect("has_wood", false)

439            .with_cost(8.0),

443/// Convert GOAP action sequence to ActionSteps.
444#[cfg(feature = "ai-goap")]
445fn goap_actions_to_steps(actions: &[GoapAction]) -> Result<Vec<ActionStep>> {
446    let mut steps = Vec::new();
447
448    for action in actions {
449        let step = match action.name.as_str() {
450            "GoToTree" => ActionStep::MoveTo {
451                x: 10,
452                y: 10,
453                speed: None,
454            },
455            "ChopWood" => ActionStep::MoveTo {
456                x: 10,
457                y: 10,
458                speed: None,
459            },
460            "GoToCampfire" => ActionStep::MoveTo {
461                x: 5,
462                y: 5,
463                speed: None,
464            },
465            "CookFood" => ActionStep::MoveTo {
466                x: 5,
467                y: 5,
468                speed: None,
469            },
470            _ => {
471                anyhow::bail!("Unknown GOAP action: {}", action.name)
472            }
473        };
474        steps.push(step);
475    }
476
477    Ok(steps)
478}
