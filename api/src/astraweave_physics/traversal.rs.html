1//! Tick-based traversal.
2//!
3//! One tick is [`TICK_DT`] (60 Hz). The sim owns displacement. Callers pass a
4//! [`ProbeFrame`] from Rapier, or a flat plane for the replay gate. Slope and
5//! jump roots go through [`crate::detmath`] (`libm`). Random draws, when a
6//! later phase needs them, use `astraweave_core::SimRng` and
7//! `SIM_SYSTEM_TRAVERSAL`. This module does not read the clock.
8

199/// Advance one tick. `world_seed` and `tick` are recorded for the replay bytes
200/// and for a future `SimRng` draw. Phase 0 does not consume a random sample.
201pub fn step(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame, tick: u64) {
202    let _ = tick;
203    if body.jump_buffer > 0 {
204        body.jump_buffer -= 1;
205    }
206    if input.jump {
207        body.jump_buffer = JUMP_BUFFER_TICKS;
208    }
209
210    if body.mode == TraversalMode::Roll {
211        step_roll(body, probe);
212        return;
213    }
214    if is_special(body.mode) {
215        step_special(body, input, probe);
216        return;
217    }
218    if try_begin_special(body, input, probe) {
219        return;
220    }
221
222    let supported = supported(body, probe);
223    if supported {
224        body.coyote = COYOTE_TICKS;
225        body.jumps_used = 0;
226        if body.mode == TraversalMode::Air {
227            body.mode = TraversalMode::Ground;
228        }
229    } else if body.coyote > 0 {
230        body.coyote -= 1;
231    }
232
233    let can_jump = supported || body.coyote > 0;
234    if body.mode == TraversalMode::Air && input.jump && body.jumps_used == 1 {
235        body.vel.y = jump_speed();
236        body.jumps_used = 2;
237        body.jump_buffer = 0;
238    } else if body.jump_buffer > 0 && can_jump && body.jumps_used == 0 {
239        body.vel.y = jump_speed();
240        body.jump_buffer = 0;
241        body.coyote = 0;
242        body.jumps_used = 1;
243        body.mode = TraversalMode::Air;
244    }
245
246    if body.mode != TraversalMode::Roll && input.roll && supported && body.stamina >= ROLL_COST {
247        begin_roll(body, input);
248        step_roll(body, probe);
249        return;
250    }
251
252    let wish = wish_dir(input);
253    let moving = wish.length_squared() > 1e-8;
254    body.crouching = input.crouch && supported && body.mode != TraversalMode::Air;
255    body.sprinting = input.sprint && moving && body.stamina > 0.0 && !body.crouching && supported;
256    body.walking = input.walk && moving && !body.sprinting && !body.crouching;
257    let speed = if !moving {
258        0.0
259    } else if body.crouching {
260        CROUCH_SPEED
261    } else if body.sprinting {
262        SPRINT_SPEED
263    } else if body.walking {
264        WALK_SPEED
265    } else {
266        RUN_SPEED
267    };
268    let planar = if body.mode == TraversalMode::Air {
269        scale(wish, AIR_CONTROL * speed)
270    } else {
271        scale(wish, speed)
272    };
273    body.vel.x = planar.x;
274    body.vel.z = planar.z;
275    if moving {
276        body.facing_yaw = detmath::atan2(wish.z, wish.x);
277    }
278
279    if supported && body.mode != TraversalMode::Air && body.mode != TraversalMode::Roll {
280        body.mode = if body.crouching {
281            TraversalMode::Crouch
282        } else {
283            TraversalMode::Ground
284        };
285    } else if !supported && body.mode != TraversalMode::Roll {
286        body.mode = TraversalMode::Air;
287    }
288
289    if body.sprinting {
290        body.stamina = (body.stamina - SPRINT_DRAIN_PER_TICK).max(0.0);
291    } else if supported && body.mode != TraversalMode::Roll {
292        body.stamina = (body.stamina + STAMINA_REGEN_PER_TICK).min(body.stamina_max);
293    }
294
295    integrate(body, probe);
296}

298fn is_special(mode: TraversalMode) -> bool {
299    matches!(
300        mode,
301        TraversalMode::Vault
302            | TraversalMode::Mantle
303            | TraversalMode::Climb
304            | TraversalMode::Swim
305            | TraversalMode::Dive
306            | TraversalMode::Slide
307            | TraversalMode::Prone
308            | TraversalMode::WallRun
309            | TraversalMode::Glide
310            | TraversalMode::LedgeHang
311            | TraversalMode::Shimmy
312    )
313}

315fn try_begin_special(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) -> bool {
316    let moving = input.move_x.abs() + input.move_y.abs() > 0.1;
317    if let Some(to) = probe.mantle_to {
318        if moving && supported(body, probe) {
319            begin_arc(body, TraversalMode::Mantle, to);
320            return true;
321        }
322    }
323    if let Some(to) = probe.vault_to {
324        if moving {
325            begin_arc(body, TraversalMode::Vault, to);
326            return true;
327        }
328    }
329    if (input.climb || input.jump) && begin_climb(body, probe) {
330        return true;
331    }
332    if let Some(surface) = probe.water_surface {
333        let depth = surface - body.pos.y;
334        if depth >= SWIM_DEPTH {
335            body.water_hold = body.water_hold.saturating_add(1);
336        } else if body.water_hold > 0 {
337            body.water_hold -= 1;
338        }
339        if body.water_hold >= WATER_HYSTERESIS && depth >= SWIM_DEPTH {
340            body.mode = if input.dive || depth >= DIVE_DEPTH {
341                TraversalMode::Dive
342            } else {
343                TraversalMode::Swim
344            };
345            body.vel = Vec3::ZERO;
346            return true;
347        }
348    }
349    if input.glide && body.mode == TraversalMode::Air && body.stamina > 0.0 {
350        body.mode = TraversalMode::Glide;
351        step_glide(body, input, probe);
352        return true;
353    }
354    if input.prone && supported(body, probe) && !probe.ceiling_blocked {
355        body.mode = TraversalMode::Prone;
356        return true;
357    }
358    if input.crouch && input.sprint && supported(body, probe) && moving {
359        body.mode = TraversalMode::Slide;
360        body.slide_speed = SPRINT_SPEED;
361        return true;
362    }
363    if probe.normal_y < 0.5 && supported(body, probe) && moving {
364        body.mode = TraversalMode::Slide;
365        body.slide_speed = RUN_SPEED;
366        return true;
367    }
368    if let Some(normal) = probe.wall_normal {
369        let speed2 = body.vel.x * body.vel.x + body.vel.z * body.vel.z;
370        if body.mode == TraversalMode::Air && speed2 > 16.0 {
371            body.mode = TraversalMode::WallRun;
372            body.climb_normal = normal;
373            body.wall_tick = 0;
374            return true;
375        }
376    }
377    false
378}

380fn begin_arc(body: &mut TraversalBody, mode: TraversalMode, to: Vec3) {
381    body.action_from = body.pos;
382    body.action_to = to;
383    body.action_tick = 0;
384    body.mode = mode;
385    body.vel = Vec3::ZERO;
386}

388fn begin_climb(body: &mut TraversalBody, probe: &ProbeFrame) -> bool {
389    if probe.climbability == Climbability::NonClimbable || body.stamina <= 0.0 {
390        return false;
391    }
392    let (point, normal) = if let Some(point) = probe.markup_point {
393        (point, probe.climb_normal.unwrap_or(Vec3::Y))
394    } else if let Some(point) = probe.climb_point {
395        (
396            point,
397            probe.climb_normal.unwrap_or(Vec3::new(0.0, 0.0, 1.0)),
398        )
399    } else {
400        return false;
401    };
402    if normal.y < OVERHANG_MIN_NORMAL_Y {
403        return false;
404    }
405    body.mode = TraversalMode::Climb;
406    body.action_to = point;
407    body.climb_normal = normal;
408    body.pos = point;
409    body.vel = Vec3::ZERO;
410    true
411}

413fn step_special(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
414    match body.mode {
415        TraversalMode::Vault => step_arc(body, VAULT_TICKS, 0.35),
416        TraversalMode::Mantle => step_arc(body, MANTLE_TICKS, 0.55),
417        TraversalMode::Climb => step_climb(body, input, probe),
418        TraversalMode::LedgeHang => step_ledge(body, input, probe),
419        TraversalMode::Shimmy => step_shimmy(body, input, probe),
420        TraversalMode::Swim => step_swim(body, input, probe, false),
421        TraversalMode::Dive => step_swim(body, input, probe, true),
422        TraversalMode::Slide => step_slide(body, input, probe),
423        TraversalMode::Prone => step_prone(body, input, probe),
424        TraversalMode::WallRun => step_wallrun(body, input, probe),
425        TraversalMode::Glide => step_glide(body, input, probe),
426        _ => {}
427    }
428}

430fn step_arc(body: &mut TraversalBody, duration: u16, arch: f32) {
431    body.action_tick = body.action_tick.saturating_add(1);
432    let t = (f32::from(body.action_tick) / f32::from(duration)).min(1.0);
433    let base = body.action_from + (body.action_to - body.action_from) * t;
434    let bump = detmath::sin(t * std::f32::consts::PI) * arch;
435    body.pos = Vec3::new(base.x, base.y + bump, base.z);
436    if body.action_tick >= duration {
437        body.pos = body.action_to;
438        body.vel = Vec3::ZERO;
439        body.mode = TraversalMode::Ground;
440        body.action_tick = 0;
441    }
442}

444fn step_climb(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
445    if probe.climbability == Climbability::NonClimbable || body.stamina <= 0.0 {
446        body.mode = TraversalMode::Air;
447        body.vel = Vec3::ZERO;
448        return;
449    }
450    let before = body.stamina;
451    let mut drain = CLIMB_DRAIN_PER_TICK;
452    if probe.climbability == Climbability::Slippery {
453        drain *= 1.5;
454    }
455    drain *= 1.0 + body.wetness.clamp(0.0, 1.0);
456    body.stamina = (body.stamina - drain).max(0.0);
457    let _ = before;
458    let wish = wish_dir(input);
459    let mut delta = scale(wish, CLIMB_SPEED * TICK_DT);
460    if probe.climbability == Climbability::Slippery {
461        delta.y -= 0.4 * TICK_DT;
462    }
463    body.pos += delta;
464    if let Some(normal) = probe.climb_normal {
465        body.climb_normal = normal;
466        if normal.y < OVERHANG_MIN_NORMAL_Y {
467            body.mode = TraversalMode::Air;
468            return;
469        }
470    }
471    if let Some(ledge) = probe.ledge_point {
472        body.mode = TraversalMode::LedgeHang;
473        body.pos = ledge;
474        body.vel = Vec3::ZERO;
475    }
476    if body.stamina <= 0.0 {
477        body.mode = TraversalMode::Air;
478    }
479}

481fn step_ledge(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
482    if probe.ledge_point.is_none() {
483        body.mode = TraversalMode::Air;
484        return;
485    }
486    body.pos = probe.ledge_point.unwrap_or(body.pos);
487    if input.move_x.abs() > 0.1 {
488        body.mode = TraversalMode::Shimmy;
489    }
490}

492fn step_shimmy(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
493    if probe.ledge_point.is_none() {
494        body.mode = TraversalMode::Air;
495        return;
496    }
497    body.pos.x += input.move_x * 1.2 * TICK_DT;
498    body.pos.y = probe.ledge_point.unwrap_or(body.pos).y;
499    if input.move_x.abs() <= 0.1 {
500        body.mode = TraversalMode::LedgeHang;
501    }
502}

504fn step_swim(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame, diving: bool) {
505    let surface = probe.water_surface.unwrap_or(body.pos.y);
506    let depth = surface - body.pos.y;
507    if depth < SWIM_DEPTH - 0.2 && body.water_hold == 0 {
508        body.mode = TraversalMode::Ground;
509        return;
510    }
511    if depth < SWIM_DEPTH {
512        if body.water_hold > 0 {
513            body.water_hold -= 1;
514        }
515        if body.water_hold == 0 && depth < SWIM_DEPTH - 0.05 {
516            body.mode = TraversalMode::Ground;
517            return;
518        }
519    }
520    let wish = wish_dir(input);
521    let speed = if diving { 2.2 } else { 2.8 };
522    body.vel.x = wish.x * speed;
523    body.vel.z = wish.z * speed;
524    body.pos.x += body.vel.x * TICK_DT;
525    body.pos.z += body.vel.z * TICK_DT;
526    if diving || input.dive || depth >= DIVE_DEPTH {
527        body.mode = TraversalMode::Dive;
528        let vertical = if input.jump { 1.5 } else { -1.2 };
529        body.vel.y = vertical;
530        body.pos.y += body.vel.y * TICK_DT;
531    } else {
532        body.mode = TraversalMode::Swim;
533        body.pos.y = surface - 0.2;
534        body.vel.y = 0.0;
535    }
536}

538fn step_slide(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
539    if input.jump {
540        body.mode = TraversalMode::Air;
541        body.vel.x = detmath::cos(body.facing_yaw) * body.slide_speed;
542        body.vel.z = detmath::sin(body.facing_yaw) * body.slide_speed;
543        body.vel.y = jump_speed() * 0.5;
544        return;
545    }
546    let downhill = if probe.normal_y < 0.85 { 2.0 } else { 0.0 };
547    body.slide_speed = (body.slide_speed + downhill * TICK_DT - SLIDE_FRICTION * TICK_DT).max(0.0);
548    body.pos.x += detmath::cos(body.facing_yaw) * body.slide_speed * TICK_DT;
549    body.pos.z += detmath::sin(body.facing_yaw) * body.slide_speed * TICK_DT;
550    if probe.support {
551        body.pos.y = floor_y(probe);
552    }
553    if body.slide_speed < 0.5 && probe.normal_y >= 0.85 {
554        body.mode = TraversalMode::Ground;
555        body.vel = Vec3::ZERO;
556    }
557}

559fn step_prone(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
560    if !input.prone && !probe.ceiling_blocked {
561        body.mode = TraversalMode::Crouch;
562        return;
563    }
564    let wish = wish_dir(input);
565    body.vel.x = wish.x * 0.8;
566    body.vel.z = wish.z * 0.8;
567    body.pos.x += body.vel.x * TICK_DT;
568    body.pos.z += body.vel.z * TICK_DT;
569    if probe.support {
570        body.pos.y = floor_y(probe);
571    }
572}

574fn step_wallrun(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
575    body.wall_tick = body.wall_tick.saturating_add(1);
576    if probe.wall_normal.is_none() || body.wall_tick > WALLRUN_TICKS {
577        body.mode = TraversalMode::Air;
578        return;
579    }
580    let wish = wish_dir(input);
581    let along = if wish.length_squared() <= 1e-8 {
582        Vec3::new(-body.climb_normal.z, 0.0, body.climb_normal.x)
583    } else {
584        wish
585    };
586    body.vel.x = along.x * WALLRUN_SPEED;
587    body.vel.z = along.z * WALLRUN_SPEED;
588    body.vel.y = -1.0;
589    body.pos.x += body.vel.x * TICK_DT;
590    body.pos.y += body.vel.y * TICK_DT;
591    body.pos.z += body.vel.z * TICK_DT;
592}

594fn step_glide(body: &mut TraversalBody, input: TraversalInput, probe: &ProbeFrame) {
595    if !input.glide || body.stamina <= 0.0 {
596        body.mode = TraversalMode::Air;
597        return;
598    }
599    body.stamina = (body.stamina - GLIDE_DRAIN_PER_TICK).max(0.0);
600    let wish = wish_dir(input);
601    body.vel.x = wish.x * 6.0;
602    body.vel.z = wish.z * 6.0;
603    body.vel.y = if probe.updraft { 2.0 } else { -GLIDE_SINK };
604    body.pos += body.vel * TICK_DT;
605    if body.stamina <= 0.0 {
606        body.mode = TraversalMode::Air;
607    }
608}

610fn begin_roll(body: &mut TraversalBody, input: TraversalInput) {
611    body.stamina = (body.stamina - ROLL_COST).max(0.0);
612    body.mode = TraversalMode::Roll;
613    body.roll_tick = 0;
614    body.invulnerable = false;
615    let mut dir = wish_dir(input);
616    if dir == Vec3::ZERO {
617        dir = Vec3::new(
618            detmath::cos(body.facing_yaw),
619            0.0,
620            detmath::sin(body.facing_yaw),
621        );
622    }
623    body.roll_dir = dir;
624}
625
626fn step_roll(body: &mut TraversalBody, probe: &ProbeFrame) {
627    if body.roll_tick >= ROLL_TICKS {
628        body.mode = TraversalMode::Ground;
629        body.invulnerable = false;
630        body.roll_tick = 0;
631        body.vel.x = 0.0;
632        body.vel.z = 0.0;
633        integrate(body, probe);
634        return;
635    }
636    body.invulnerable = body.roll_tick >= ROLL_IFRAME_START && body.roll_tick <= ROLL_IFRAME_END;
637    body.vel.x = body.roll_dir.x * ROLL_SPEED;
638    body.vel.y = 0.0;
639    body.vel.z = body.roll_dir.z * ROLL_SPEED;
640    body.pos.x += body.vel.x * TICK_DT;
641    body.pos.z += body.vel.z * TICK_DT;
642    if probe.support {
643        body.pos.y = floor_y(probe);
644    }
645    body.facing_yaw = detmath::atan2(body.roll_dir.z, body.roll_dir.x);
646    body.roll_tick += 1;
647}
648
649fn integrate(body: &mut TraversalBody, probe: &ProbeFrame) {
650    body.vel.y -= GRAVITY * TICK_DT;
651    body.pos.x += body.vel.x * TICK_DT;
652    body.pos.y += body.vel.y * TICK_DT;
653    body.pos.z += body.vel.z * TICK_DT;
654    if lands(body, probe) {
655        body.pos.y = floor_y(probe);
656        body.vel.y = 0.0;
657        if body.mode == TraversalMode::Air {
658            body.mode = TraversalMode::Ground;
659            body.coyote = COYOTE_TICKS;
660            body.jumps_used = 0;
661        }
662    }
663}
664
665fn floor_y(probe: &ProbeFrame) -> f32 {
666    match probe.step_height {
667        Some(height) if height > 0.0 && height <= MAX_STEP => probe.ground_y + height,
668        _ => probe.ground_y,
669    }
670}
671
672fn lands(body: &TraversalBody, probe: &ProbeFrame) -> bool {
673    probe.support && body.vel.y <= 0.0 && body.pos.y <= floor_y(probe) + SNAP
674}
675
676fn supported(body: &TraversalBody, probe: &ProbeFrame) -> bool {
677    probe.support && body.vel.y <= 0.0 && body.pos.y <= floor_y(probe) + SNAP
678}
679
680fn jump_speed() -> f32 {
681    detmath::sqrt(2.0 * GRAVITY * JUMP_HEIGHT)
682}
683
684fn wish_dir(input: TraversalInput) -> Vec3 {
685    let forward = Vec3::new(detmath::cos(input.yaw), 0.0, detmath::sin(input.yaw));
686    let right = Vec3::new(-forward.z, 0.0, forward.x);
687    let wish = Vec3::new(
688        forward.x * input.move_y + right.x * input.move_x,
689        0.0,
690        forward.z * input.move_y + right.z * input.move_x,
691    );
692    normalize_xz(wish)
693}
694
695fn normalize_xz(v: Vec3) -> Vec3 {
696    let len2 = v.x * v.x + v.z * v.z;
697    if len2 <= 1e-8 {
698        return Vec3::ZERO;
699    }
700    let inv = 1.0 / detmath::sqrt(len2);
701    Vec3::new(v.x * inv, 0.0, v.z * inv)
702}
703
704fn scale(v: Vec3, s: f32) -> Vec3 {
705    Vec3::new(v.x * s, v.y * s, v.z * s)
706}

708/// Largest gap from a render-mesh sample to the nearest collider sample.
709/// Climbable assets must stay within [`COLLISION_DEVIATION_BUDGET`].
710pub fn max_collider_deviation(render: &[Vec3], collider: &[Vec3]) -> f32 {
711    let mut max = 0.0f32;
712    for point in render {
713        let mut best = f32::MAX;
714        for sample in collider {
715            let dx = point.x - sample.x;
716            let dy = point.y - sample.y;
717            let dz = point.z - sample.z;
718            let dist = detmath::sqrt(dx * dx + dy * dy + dz * dz);
719            if dist < best {
720                best = dist;
721            }
722        }
723        if best > max {
724            max = best;
725        }
726    }
727    max
728}
729
730pub fn climb_collision_within_budget(render: &[Vec3], collider: &[Vec3]) -> bool {
731    max_collider_deviation(render, collider) <= COLLISION_DEVIATION_BUDGET
732}

734/// 180 ticks of running, one jump, and one roll on a flat plane.
735///
736/// The bytes include position, velocity, stamina, and the roll tick. Two
737/// calls return the same `Vec`. A constant digest cannot satisfy the asserts
738/// in the replay test.
739pub fn replay_bytes() -> Vec<u8> {
740    let mut body = TraversalBody::default();
741    let probe = ProbeFrame::plane(0.0);
742    let mut out = Vec::with_capacity(180 * 48);
743    push_u64(&mut out, REPLAY_SEED);
744    push_u32(&mut out, REPLAY_TICKS);
745    for tick in 0..REPLAY_TICKS {
746        let input = replay_input(tick);
747        step(&mut body, input, &probe, u64::from(tick));
748        encode_body(&mut out, tick, &body);
749    }
750    out
751}
752
753fn replay_input(tick: u32) -> TraversalInput {
754    let running = tick < 40 || (110..130).contains(&tick);
755    let jumping = tick == 40;
756    let rolling = tick == 130;
757    let moving = running || jumping || (130..160).contains(&tick);
758    TraversalInput {
759        move_x: 0.0,
760        move_y: if moving { 1.0 } else { 0.0 },
761        yaw: 0.0,
762        jump: jumping,
763        roll: rolling,
764        sprint: false,
765        crouch: false,
766        walk: false,
767        climb: false,
768        glide: false,
769        dive: false,
770        prone: false,
771    }
772}
773
774fn encode_body(out: &mut Vec<u8>, tick: u32, body: &TraversalBody) {
775    push_u32(out, tick);
776    out.push(body.mode as u8);
777    push_f32(out, body.pos.x);
778    push_f32(out, body.pos.y);
779    push_f32(out, body.pos.z);
780    push_f32(out, body.vel.x);
781    push_f32(out, body.vel.y);
782    push_f32(out, body.vel.z);
783    push_f32(out, body.stamina);
784    push_u16(out, body.roll_tick);
785    out.push(u8::from(body.invulnerable));
786    out.push(body.jumps_used);
787}
788
789fn push_u16(out: &mut Vec<u8>, value: u16) {
790    out.extend_from_slice(&value.to_le_bytes());
791}
792
793fn push_u32(out: &mut Vec<u8>, value: u32) {
794    out.extend_from_slice(&value.to_le_bytes());
795}
796
797fn push_u64(out: &mut Vec<u8>, value: u64) {
798    out.extend_from_slice(&value.to_le_bytes());
799}
800
801fn push_f32(out: &mut Vec<u8>, value: f32) {
802    out.extend_from_slice(&value.to_bits().to_le_bytes());
803}
