6/// Model text above this is rejected before `serde_json` allocates a document.
7pub const MAX_MODEL_JSON_BYTES: usize = 64 * 1024;

9/// Hard ceiling for any capped body read, including embedding batches.
10pub const MAX_MODEL_BODY_LIMIT: usize = 8 * 1024 * 1024;

12/// Raw NDJSON budget. Each token repeats the envelope, so this is larger
13/// than the 64 KiB extracted-content cap and still under [`MAX_MODEL_BODY_LIMIT`].
14pub const MAX_MODEL_STREAM_BYTES: usize = MAX_MODEL_JSON_BYTES * 16;

16/// The prefix of `text` that fits in [`MAX_MODEL_JSON_BYTES`].
17pub fn bounded_model_text(text: &str) -> &str {
18    if text.len() <= MAX_MODEL_JSON_BYTES {
19        return text;
20    }
21    let mut end = MAX_MODEL_JSON_BYTES;
22    while end > 0 && !text.is_char_boundary(end) {
23        end -= 1;
24    }
25    &text[..end]
26}

28/// Count `extra` toward the raw NDJSON budget.
29///
30/// Empty lines and invalid UTF-8 still count. Extracted content is capped
31/// separately at [`MAX_MODEL_JSON_BYTES`].
32///
33/// The budget is inclusive. Exactly [`MAX_MODEL_STREAM_BYTES`] is kept, and
34/// one more byte fails. [`append_capped`] uses the same `>` check.
35#[cfg(feature = "ollama")]
36pub fn note_stream_bytes(seen: &mut usize, extra: usize) -> Result<()> {
37    *seen = seen.saturating_add(extra);
38    if *seen > MAX_MODEL_STREAM_BYTES {
39        bail!(
40            "model stream exceeds {MAX_MODEL_STREAM_BYTES} byte cap ({} bytes)",
41            *seen
42        );
43    }
44    Ok(())
45}

47/// Keep `text` when it fits in the 64 KiB content cap.
48///
49/// Exactly [`MAX_MODEL_JSON_BYTES`] is returned. One more byte is an error.
50#[cfg(feature = "ollama")]
51fn finish_capped(text: String) -> Result<String> {
52    if text.len() > MAX_MODEL_JSON_BYTES {
53        bail!("model text exceeds 64 KiB cap ({} bytes)", text.len());
54    }
55    Ok(text)

58#[cfg(feature = "ollama")]
59fn append_capped(dst: &mut String, extra: &str) -> Result<()> {
60    let next = dst.len().saturating_add(extra.len());
61    if next > MAX_MODEL_JSON_BYTES {
62        bail!("model text exceeds 64 KiB cap ({next} bytes)");
63    }
64    dst.push_str(extra);
65    Ok(())
66}

68/// Take one NDJSON line from the default chat stream.
69#[cfg(feature = "ollama")]
70fn push_stream_line(assistant_acc: &mut String, line: &str, done_flag: &mut bool) -> Result<()> {
71    if let Ok(v) = serde_json::from_str::<serde_json::Value>(line) {
72        if let Some(content) = v
73            .get("message")
74            .and_then(|msg| msg.get("content"))
75            .and_then(|c| c.as_str())
76        {
77            append_capped(assistant_acc, content)?;
78        }
79        if let Some(resp_txt) = v.get("response").and_then(|r| r.as_str()) {
80            append_capped(assistant_acc, resp_txt)?;
81        }
82        if let Some(choices) = v.get("choices").and_then(|c| c.as_array()) {
83            if let Some(first) = choices.first() {
84                if let Some(content) = first
85                    .get("message")
86                    .and_then(|msg| msg.get("content"))
87                    .and_then(|c| c.as_str())
88                {
89                    append_capped(assistant_acc, content)?;
90                }
91            }
92        }
93        if v.get("done").and_then(|d| d.as_bool()) == Some(true) {
94            *done_flag = true;
95        }
96    } else {
97        append_capped(assistant_acc, line)?;
98    }
99    Ok(())
100}

117/// Read a body until `limit` bytes, then stop.
118#[cfg(feature = "ollama")]
119pub async fn read_capped_text_limited(response: reqwest::Response, limit: usize) -> Result<String> {
120    use futures_util::StreamExt;
121    accept_body_limit(limit)?;
122    let mut stream = response.bytes_stream();
123    let mut buf = Vec::new();
124    while let Some(chunk) = stream.next().await {
125        let chunk = chunk.context("model body read failed")?;
126        let Some(next) = buf.len().checked_add(chunk.len()) else {
127            bail!("model body length overflow");
128        };
129        if next > limit {
130            if limit == MAX_MODEL_JSON_BYTES {
131                bail!("model text exceeds 64 KiB cap ({next} bytes)");
132            }
133            bail!("model text exceeds {limit} byte cap ({next} bytes)");
134        }
135        buf.extend_from_slice(&chunk);
136    }
137    String::from_utf8(buf).context("model body is not utf-8")
138}

140/// How long the next body read may block, and which deadline a timeout is.
141struct ReadWait {
142    wait: std::time::Duration,
143    on_timeout: &'static str,
144}
145
146/// Silence, the current gap, and the stream lifetime.
147///
148/// `idle` is the silence limit. `overall` caps the same gap: when it is
149/// longer than `idle`, the wait is `idle`, so a caller that passes twice
150/// the idle limit does not get two idle periods. `lifetime` is measured
151/// from `life_started` and does not move when a chunk arrives.
152/// `idle_timeout` is the silence error. The other errors are
153/// `streaming overall deadline` and `streaming lifetime deadline`.
154#[cfg(feature = "ollama")]
155fn next_read_wait(
156    idle: std::time::Duration,
157    overall: std::time::Duration,
158    gap_started: std::time::Instant,
159    lifetime: std::time::Duration,
160    life_started: std::time::Instant,
161    idle_timeout: &'static str,
162) -> Result<ReadWait> {
163    let gap_left = overall.saturating_sub(gap_started.elapsed());
164    let life_left = lifetime.saturating_sub(life_started.elapsed());
165    if life_left.is_zero() {
166        bail!("streaming lifetime deadline");
167    }
168    if gap_left.is_zero() {
169        bail!("streaming overall deadline");
170    }
171    let wait = idle.min(gap_left).min(life_left);
172    let on_timeout = if life_left <= gap_left && life_left < idle {
173        "streaming lifetime deadline"
174    } else if wait < idle {
175        "streaming overall deadline"
176    } else {
177        idle_timeout
178    };
179    Ok(ReadWait { wait, on_timeout })
180}

193/// Read `inner` until silence, the current gap, or `lifetime` ends.
194///
195/// Each read waits the silence limit, the time still left in the current
196/// gap, or the time still left in `lifetime`, whichever is shortest.
197/// `lifetime` starts when this stream is built and does not reset. A
198/// non-empty chunk starts a new gap, so late wakes whose gaps stay inside
199/// `idle` are kept. An empty chunk does not start a gap: an HTTP/2 DATA
200/// frame with no bytes must not postpone the deadline. The first error is
201/// yielded once. The next poll ends the stream.
202#[cfg(feature = "ollama")]
203pub fn bytes_within_deadline<S, T, E>(
204    inner: S,
205    idle: std::time::Duration,
206    overall: std::time::Duration,
207    lifetime: std::time::Duration,
208) -> impl futures_util::Stream<Item = Result<T>> + Send
209where
210    S: futures_util::Stream<Item = Result<T, E>> + Unpin + Send + 'static,
211    T: AsRef<[u8]> + Send + 'static,
212    E: std::error::Error + Send + Sync + 'static,
213{
214    use futures_util::StreamExt;
215    futures_util::stream::unfold(
216        (
217            inner,
218            false,
219            std::time::Instant::now(),
220            std::time::Instant::now(),
221        ),
222        move |(mut inner, stop, gap_started, life_started)| async move {
223            if stop {
224                return None;
225            }
226            let decision = match next_read_wait(
227                idle,
228                overall,
229                gap_started,
230                lifetime,
231                life_started,
232                "stream idle timeout",
233            ) {
234                Ok(decision) => decision,
235                Err(err) => {
236                    return Some((Err(err), (inner, true, gap_started, life_started)));
237                }
238            };
239            match tokio::time::timeout(decision.wait, inner.next()).await {
240                Ok(Some(Ok(chunk))) => {
241                    let gap_started = if chunk.as_ref().is_empty() {
242                        gap_started
243                    } else {
244                        std::time::Instant::now()
245                    };
246                    Some((Ok(chunk), (inner, false, gap_started, life_started)))
247                }
248                Ok(Some(Err(err))) => Some((
249                    Err(anyhow::Error::new(err).context("stream body error")),
250                    (inner, true, gap_started, life_started),
251                )),
252                Ok(None) => None,
253                Err(_) => Some((
254                    Err(anyhow::anyhow!(decision.on_timeout)),
255                    (inner, true, gap_started, life_started),
256                )),
257            }
258        },
259    )
260}

262/// Split a drained NDJSON block into lines.
263///
264/// Lines before a bad byte are kept. That bad line is an error, and lines
265/// after it in the same slice are not decoded.
266#[cfg(feature = "ollama")]
267pub(crate) fn ndjson_line_strings(bytes: &[u8]) -> Vec<Result<String>> {
268    let mut out = Vec::new();
269    for mut piece in bytes.split(|b| *b == b'\n') {
270        if piece.last() == Some(&b'\r') {
271            piece = &piece[..piece.len() - 1];
272        }
273        if piece.is_empty() {
274            continue;
275        }
276        match String::from_utf8(piece.to_vec()) {
277            Ok(line) => out.push(Ok(line)),
278            Err(_) => {
279                out.push(Err(anyhow::anyhow!("model stream line is not utf-8")));
280                break;
281            }
282        }
283    }
284    out
285}

294/// Scan NDJSON bytes until one error or until `on_line` says to stop.
295///
296/// Each read waits the idle limit, the time still left in the gap since
297/// the last non-empty chunk, or the time still left in `lifetime`,
298/// whichever is shortest. `lifetime` starts when this scan is built.
299/// A non-empty chunk starts a new gap. An empty chunk does not. A bad
300/// line does not fail earlier lines from the same read. After one error
301/// the next poll ends the stream, so a later chunk is not read and the
302/// byte cap still applies up to the error.
303#[cfg(feature = "ollama")]
304pub(crate) fn scan_model_ndjson<S, T, E, F>(
305    inner: S,
306    idle: std::time::Duration,
307    overall: std::time::Duration,
308    lifetime: std::time::Duration,
309    on_line: F,
310) -> impl futures_util::Stream<Item = Result<String>> + Send
311where
312    S: futures_util::Stream<Item = Result<T, E>> + Unpin + Send + 'static,
313    T: AsRef<[u8]> + Send + 'static,
314    E: std::error::Error + Send + Sync + 'static,
315    F: FnMut(&str) -> NdjsonYield + Send + 'static,
316{
317    use futures_util::StreamExt;
318
319    struct ScanState<B, F> {
320        bytes: B,
321        buf: Vec<u8>,
322        seen: usize,
323        pending: std::vec::IntoIter<Result<String>>,
324        stop: bool,
325        kept: usize,
326        on_line: F,
327    }
328
329    let bytes = Box::pin(bytes_within_deadline(inner, idle, overall, lifetime));
330    futures_util::stream::unfold(
331        ScanState {
332            bytes,
333            buf: Vec::new(),
334            seen: 0,
335            pending: Vec::new().into_iter(),
336            stop: false,
337            kept: 0,
338            on_line,
339        },
340        |mut state| async move {
341            loop {
342                if let Some(item) = state.pending.next() {
343                    return Some((item, state));
344                }
345                if state.stop {
346                    return None;
347                }
348                match state.bytes.next().await {
349                    None => return None,
350                    Some(Err(err)) => {
351                        state.stop = true;
352                        return Some((Err(err), state));
353                    }
354                    Some(Ok(chunk)) => {
355                        let chunk = chunk.as_ref();
356                        if let Err(err) = note_stream_bytes(&mut state.seen, chunk.len()) {
357                            state.stop = true;
358                            return Some((Err(err), state));
359                        }
360                        state.buf.extend_from_slice(chunk);
361                        let mut batch = Vec::new();
362                        let mut stop_now = false;
363                        if let Some(last_nl) = state.buf.iter().rposition(|&b| b == b'\n') {
364                            let complete: Vec<u8> = state.buf.drain(..=last_nl).collect();
365                            for line in ndjson_line_strings(&complete) {
366                                match line {
367                                    Err(err) => {
368                                        batch.push(Err(err));
369                                        stop_now = true;
370                                        break;
371                                    }
372                                    Ok(line) => {
373                                        let trimmed = line.trim();
374                                        if trimmed.is_empty() {
375                                            continue;
376                                        }
377                                        let yielded = (state.on_line)(trimmed);
378                                        for piece in yielded.pieces {
379                                            if piece.is_empty() {
380                                                continue;
381                                            }
382                                            let next = state.kept.saturating_add(piece.len());
383                                            if next > MAX_MODEL_JSON_BYTES {
384                                                batch.push(Err(anyhow::anyhow!(
385                                                    "model text exceeds 64 KiB cap ({next} bytes)"
386                                                )));
387                                                stop_now = true;
388                                                break;
389                                            }
390                                            state.kept = next;
391                                            batch.push(Ok(piece));
392                                        }
393                                        if stop_now || yielded.stop {
394                                            stop_now = true;
395                                            break;
396                                        }
397                                    }
398                                }
399                            }
400                        }
401                        state.pending = batch.into_iter();
402                        if stop_now {
403                            state.stop = true;
404                        }
405                    }
406                }
407            }
408        },
409    )
410}

885/// Keep a loopback model URL.
886///
887/// A URL is kept only when it is `http` or `https` and has no userinfo. Userinfo,
888/// a non-http scheme, and a string the parser rejects are always replaced with
889/// `http://127.0.0.1:11434`, including when `allow_remote` is set. A non-loopback
890/// host is kept only when `allow_remote` is set. Otherwise the host must be
891/// `127.0.0.0/8`, `::1`, or `localhost`.
892///
893/// Every loopback URL check in the workspace goes through [`is_loopback_http_url`].
894pub fn loopback_model_url(url: String, allow_remote: bool) -> String {
895    const FALLBACK: &str = "http://127.0.0.1:11434";
896    // Userinfo is rejected even when `allow_remote` is set. The raw string is
897    // not returned until the URL parses as http(s) with an empty userinfo.
898    if !http_url_without_userinfo(&url) {
899        return FALLBACK.to_string();
900    }
901    if allow_remote || is_loopback_http_url(&url) {
902        url
903    } else {
904        FALLBACK.to_string()
905    }
906}

908/// Keep `url` only when [`loopback_model_url`] would not rewrite it.
909///
910/// Constructors use this so a rejected host or userinfo cannot be stored, and
911/// a bearer token cannot be attached to the loopback fallback. The error does
912/// not include the URL.
913pub fn require_kept_model_url(url: String, allow_remote: bool) -> Result<String> {
914    let kept = loopback_model_url(url.clone(), allow_remote);
915    if kept != url {
916        bail!("model URL was rejected");
917    }
918    Ok(kept)
919}

943/// Builder for a model HTTP client that cannot leave the machine.
944///
945/// `no_proxy` ignores `HTTP_PROXY`, `HTTPS_PROXY`, and `ALL_PROXY`. Redirects
946/// are followed only when the next hop is still loopback. The name `localhost`
947/// is pinned to `127.0.0.1`; the URL port is kept.
948///
949/// [`allowing_remote`](OllamaClient::allowing_remote) does not use this builder.
950/// It keeps the caller URL and uses reqwest's default client, which honors
951/// those proxy variables and follows redirects.
952#[cfg(feature = "ollama")]
953pub fn loopback_http_client() -> reqwest::ClientBuilder {
954    reqwest::Client::builder()
955        .no_proxy()
956        .redirect(loopback_redirect_policy())
957        .resolve(
958            "localhost",
959            std::net::SocketAddr::from((std::net::Ipv4Addr::LOCALHOST, 0)),
960        )
961        .resolve(
962            "localhost.",
963            std::net::SocketAddr::from((std::net::Ipv4Addr::LOCALHOST, 0)),
964        )
965}

967#[cfg(feature = "ollama")]
968fn loopback_redirect_policy() -> reqwest::redirect::Policy {
969    reqwest::redirect::Policy::custom(|attempt| {
970        if attempt.previous().len() >= 8 {
971            attempt.error("too many loopback redirects")
972        } else if is_loopback_http_url(attempt.url().as_str()) {
973            attempt.follow()
974        } else {
975            attempt.error("redirect is not loopback")
976        }
977    })
978}

981/// non-loopback URL on reqwest's default client (proxy env and redirects).

983#[cfg(feature = "ollama")]

985    let remote = allow_remote && !is_loopback_http_url(url);

987        reqwest::Client::builder()

1017/// Wait `idle` for response headers. The body is not covered.
1018///
1019/// `RequestBuilder::timeout` would also cancel a chunked body. Callers that
1020/// stream after the headers use [`bytes_within_deadline`]. A call that has
1021/// a stream lifetime passes [`header_wait`] so this wait cannot outlive it.
1022#[cfg(feature = "ollama")]
1023pub async fn send_for_headers(
1024    idle: std::time::Duration,
1025    send: impl std::future::Future<Output = Result<reqwest::Response, reqwest::Error>>,
1026) -> Result<reqwest::Response> {
1027    match tokio::time::timeout(idle, send).await {
1028        Ok(Ok(response)) => Ok(response),
1029        Ok(Err(err)) => Err(anyhow::Error::new(err).context("model request")),
1030        Err(_) => Err(anyhow::anyhow!("model headers idle timeout")),
1031    }
1032}

1069/// Header wait: the idle limit, or the time still left in the stream
1070/// lifetime, whichever is shorter.
1071#[cfg(feature = "ollama")]
1072pub fn header_wait(
1073    idle: std::time::Duration,
1074    lifetime_left: std::time::Duration,
1075) -> std::time::Duration {
1076    idle.min(lifetime_left)
1077}

1079/// `default`, or a shorter `OLLAMA_STREAM_LIFETIME_MS`.
1080///
1081/// A missing, zero, non-numeric, or longer value keeps `default`. The
1082/// variable cannot extend the call.
1083#[cfg(feature = "ollama")]
1084pub fn stream_call_lifetime(default: std::time::Duration) -> std::time::Duration {
1085    let Ok(raw) = std::env::var("OLLAMA_STREAM_LIFETIME_MS") else {
1086        return default;
1087    };
1088    let Ok(ms) = raw.parse::<u64>() else {
1089        return default;
1090    };
1091    let requested = std::time::Duration::from_millis(ms);
1092    if requested.is_zero() || requested >= default {
1093        default
1094    } else {
1095        requested
1096    }
1097}

1099/// Wait for headers, stopping at `lifetime_left` when that is shorter
1100/// than `idle`.
1101///
1102/// A timeout caused by the lifetime is `streaming lifetime deadline`.
1103/// A timeout caused by `idle` stays `model headers idle timeout`.
1104#[cfg(feature = "ollama")]
1105pub async fn send_headers_for_call(
1106    idle: std::time::Duration,
1107    lifetime_left: std::time::Duration,
1108    send: impl std::future::Future<Output = Result<reqwest::Response, reqwest::Error>>,
1109) -> Result<reqwest::Response> {

1112        bail!("streaming lifetime deadline");
1113    }
1114    match send_for_headers(wait, send).await {
1115        Err(err) if wait < idle && err.to_string().contains("model headers idle timeout") => {
1116            Err(err.context("streaming lifetime deadline"))
1117        }
1118        other => other,
1119    }

1247        &self,
1248        prompt: &str,

1252        Ok(Box::pin(futures_util::stream::once(
1253            async move { Ok(result) },
1254        )))
1255    }

1548    async fn complete_with_optional_format(
1549        &self,
1550        prompt: &str,
1551        format_schema: Option<&serde_json::Value>,
1552    ) -> Result<String> {
1553        ensure_model_request(&self.url, self.allow_remote)?;
1554        let mut schema_owned = format_schema.cloned();
1555        #[derive(serde::Serialize)]
1556        struct Msg<'a> {
1557            role: &'a str,
1558            content: &'a str,
1559        }
1560        #[derive(serde::Serialize)]
1561        struct Req<'a> {
1562            model: &'a str,
1563            messages: Vec<Msg<'a>>,
1564            stream: bool,
1565            #[serde(skip_serializing_if = "Option::is_none")]
1566            format: Option<&'a serde_json::Value>,
1567        }
1568
1569        let client = self.client.clone();
1570        // Build a non-chunked JSON body string and set Connection: close so the server
1571        // will finish the response and close the connection (avoids some streaming behaviors).
1572        // Build a JSON body for non-streaming attempts. Use `.json()` on the request so
1573        // reqwest sets proper headers and Content-Length.
1574
1575        // Some Ollama setups can take a long time to load a model initially (minutes) or
1576        // will stream responses. Allow a long configurable timeout and fall back to a
1577        // streaming read if a non-stream response isn't available.
1578        let idle = stream_read_idle();
1579
1580        // First try a non-streaming request with a long timeout so we get a buffered JSON
1581        // response if the server supports it and the model responds within the window.
1582        let mut text = String::new();
1583        let chat_url = format!("{}/api/chat", self.url.trim_end_matches('/'));
1584        // Configure non-stream attempts/timeout via env with conservative defaults
1585        let max_nonstream_attempts: u32 = std::env::var("OLLAMA_NONSTREAM_ATTEMPTS")
1586            .ok()
1587            .and_then(|s| s.parse().ok())
1588            .unwrap_or(1);
1589        let nonstream_timeout_secs: u64 = std::env::var("OLLAMA_NONSTREAM_TIMEOUT_SECS")
1590            .ok()
1591            .and_then(|s| s.parse().ok())
1592            .unwrap_or(20);
1593
1594        let mut non_stream_ok = false;
1595        let mut attempt = 0u32;
1596        while attempt < max_nonstream_attempts && !non_stream_ok && !self.low_latency {
1597            attempt += 1;
1598            let schema_on_this_attempt = schema_owned.is_some();
1599            let body = Req {
1600                model: &self.model,
1601                messages: vec![Msg {
1602                    role: "user",
1603                    content: prompt,
1604                }],
1605                stream: false,
1606                format: schema_owned.as_ref(),
1607            };
1608            let backoff = std::time::Duration::from_millis(250 * (1 << (attempt - 1)));
1609            println!(
1610                "[ollama] non-stream attempt {}/{} to {} (timeout {}s)",
1611                attempt, max_nonstream_attempts, chat_url, nonstream_timeout_secs
1612            );
1613            let ns_attempt_start = std::time::Instant::now();
1614            // Use .json() so reqwest sets Content-Length and proper headers. Some servers
1615            // close the connection or reject raw bodies without length which can cause
1616            // "error sending request" failures; using .json() is more reliable here.
1617            let attempt_deadline = std::time::Duration::from_secs(nonstream_timeout_secs);
1618            enum Attempt {
1619                Text(String),
1620                Retry,
1621                SchemaRejected,
1622                Fail(anyhow::Error),
1623            }
1624            let timed = tokio::time::timeout(attempt_deadline, async {
1625                let resp = match send_for_headers(
1626                    attempt_deadline,
1627                    client
1628                        .post(&chat_url)
1629                        .header("Accept", "application/json")
1630                        .json(&body)
1631                        .send(),
1632                )
1633                .await
1634                {
1635                    Ok(resp) => resp,
1636                    Err(err) => {
1637                        println!(
1638                            "[ollama] non-streaming request failed on attempt {attempt}: {err:?}"
1639                        );
1640                        return Attempt::Retry;
1641                    }
1642                };
1643                println!(
1644                    "[ollama] non-streaming request returned status {}",
1645                    resp.status()
1646                );
1647                if resp.status().is_success() {
1648                    match read_capped_text(resp).await {
1649                        Ok(t) if !t.trim().is_empty() => Attempt::Text(t),
1650                        Ok(_) => {
1651                            println!(
1652                                "[ollama] buffered body was empty on attempt {attempt}, will retry/fallback"
1653                            );
1654                            Attempt::Retry
1655                        }
1656                        Err(err) => Attempt::Fail(err.context("Ollama non-streaming body read failed")),
1657                    }
1658                } else {
1659                    let status = resp.status();
1660                    match read_capped_text(resp).await {
1661                        Ok(b) if schema_on_this_attempt && plan_format::schema_format_rejected(status.as_u16(), &b) => {
1662                            Attempt::SchemaRejected
1663                        }
1664                        Ok(b) => Attempt::Fail(anyhow::anyhow!(
1665                            "Ollama non-streaming request returned error status {status}: {b}"
1666                        )),
1667                        Err(err) => Attempt::Fail(err.context(format!(
1668                            "Ollama non-streaming status {status} body read failed"
1669                        ))),
1670                    }
1671                }
1672            })
1673            .await;
1674            match timed {
1675                Err(_) => {
1676                    println!(
1677                        "[ollama] non-streaming attempt {attempt} timed out after {nonstream_timeout_secs}s"
1678                    );
1679                }
1680                Ok(Attempt::Text(t)) => {
1681                    println!(
1682                        "[ollama] received non-empty buffered body ({} bytes)",
1683                        t.len()
1684                    );
1685                    text = t;
1686                    non_stream_ok = true;
1687                    println!(
1688                        "[ollama] timing nonstream attempt {attempt}: total={}ms",
1689                        ns_attempt_start.elapsed().as_millis()
1690                    );
1691                }
1692                Ok(Attempt::Retry) => {}
1693                Ok(Attempt::SchemaRejected) => {
1694                    schema_owned = None;
1695                }
1696                Ok(Attempt::Fail(err)) => return Err(err),
1697            }
1698
1699            if !non_stream_ok {
1700                // small backoff before retrying
1701                tokio::time::sleep(backoff).await;
1702            }
1703        }
1704
1705        // If we didn't get a usable buffered body, or we're in low-latency mode, attempt a streaming read where we
1706        // accumulate incoming bytes and try to extract the assistant content as it arrives.
1707        if text.trim().is_empty() || self.low_latency {
1708            // Build a streaming request (request the server to stream if it supports it)
1709            let mut send_schema = schema_owned.clone();
1710            let mut stream_body = serde_json::json!({
1711                "model": &self.model,
1712                "messages": [{ "role": "user", "content": prompt }],
1713                "stream": true,
1714                "options": { "temperature": 0.1, "num_predict": 512 }
1715            });
1716            plan_format::apply_chat_format(
1717                &mut stream_body,
1718                send_schema.as_ref(),
1719                self.force_format_json,
1720            );
1721            if let Some(ka) = &self.keep_alive {
1722                stream_body["keep_alive"] = serde_json::json!(ka);
1723            }
1724
1725            // The stored client set `connect_timeout` in `build_model_client`.
1726            // Header waits and the body share this instant. Each header wait
1727            // is the idle limit or the time still left, whichever is shorter.
1728            let start = std::time::Instant::now();
1729            let lifetime = stream_call_lifetime(DEFAULT_STREAM_LIFETIME);
1730
1731            let stream_url = format!("{}/api/chat", self.url.trim_end_matches('/'));
1732            println!("[ollama] initiating streaming POST to {}", stream_url);
1733            let resp = send_headers_for_call(
1734                idle,
1735                lifetime.saturating_sub(start.elapsed()),
1736                client
1737                    .post(&stream_url)
1738                    .header("Accept", "application/json")
1739                    .header("Content-Type", "application/json")
1740                    .json(&stream_body)
1741                    .send(),
1742            )
1743            .await
1744            .context("Failed to initiate streaming request to Ollama")?;
1745
1746            println!(
1747                "[ollama] streaming request returned status {}",
1748                resp.status()
1749            );
1750
1751            // Determine which response we'll stream from: prefer /api/chat, but fall back to /api/generate
1752            let resp_for_stream: Option<reqwest::Response> = if resp.status().is_success() {
1753                Some(resp)
1754            } else {
1755                let status = resp.status();
1756                if status.as_u16() == 404 || status.as_u16() == 405 {
1757                    println!(
1758                        "[ollama] /api/chat returned {}, attempting /api/generate as fallback",
1759                        status
1760                    );
1761                    let gen_url = format!("{}/api/generate", self.url.trim_end_matches('/'));
1762                    let gen_resp = send_headers_for_call(
1763                        idle,
1764                        lifetime.saturating_sub(start.elapsed()),
1765                        client
1766                            .post(&gen_url)
1767                            .header("Accept", "application/json")
1768                            .header("Content-Type", "application/json")
1769                            .json(&stream_body)
1770                            .send(),
1771                    )
1772                    .await
1773                    .context("Failed to initiate streaming request to Ollama /api/generate")?;
1774
1775                    println!(
1776                        "[ollama] /api/generate returned status {}",
1777                        gen_resp.status()
1778                    );
1779                    if gen_resp.status().is_success() {
1780                        Some(gen_resp)
1781                    } else {
1782                        let s = gen_resp.status();
1783                        // The error body shares `start`. Stop at the time
1784                        // still left, or at 64 KiB, whichever comes first.
1785                        let b = read_capped_text_within(
1786                            gen_resp,
1787                            lifetime.saturating_sub(start.elapsed()),
1788                        )
1789                        .await
1790                        .with_context(|| {
1791                            format!("Ollama /api/generate status {s} body read failed")
1792                        })?;
1793                        if send_schema.is_some()
1794                            && plan_format::schema_format_rejected(s.as_u16(), &b)
1795                        {
1796                            let _ = send_schema.take();
1797                            if self.force_format_json {
1798                                stream_body["format"] = serde_json::json!("json");
1799                            } else if let Some(map) = stream_body.as_object_mut() {
1800                                map.remove("format");
1801                            }
1802                            let retry = send_headers_for_call(
1803                                idle,
1804                                lifetime.saturating_sub(start.elapsed()),
1805                                client

1960                        // deadline applies to the next wait, not to lines
1961                        // sitting in `raw`.

1967                            };

1971                                end_line -= 1;
1972                            }
1973                            let line_bytes = &raw[consumed..end_line];
1974                            consumed = nl + 1;
1975                            if line_bytes.is_empty() {
1976                                continue;
1977                            }
1978                            // Incomplete scalars stay in `raw` until a later chunk
1979                            // finishes the line, so a split UTF-8 character is kept.
1980                            let line = match std::str::from_utf8(line_bytes) {
1981                                Ok(line) => line,
1982                                Err(_) => {
1983                                    body_error =
1984                                        Some(anyhow::anyhow!("model stream line is not utf-8"));
1985                                    capped = true;
1986                                    break;
1987                                }
1988                            };
1989                            let line = line.trim();
1990                            if line.is_empty() {
1991                                continue;
1992                            }
1993                            if first_token_at.is_none() {
1994                                first_token_at = Some(std::time::Instant::now());
1995                            }
1996                            if let Err(err) =
1997                                push_stream_line(&mut assistant_acc, line, &mut done_flag)
1998                            {
1999                                body_error = Some(err);
2000                                capped = true;
2001                                break;
2002                            }
2003                        }
2004                        if consumed == raw.len() {
2005                            raw.clear();
2006                        } else if consumed > 0 {
2007                            raw.drain(..consumed);
2008                        }
2009                        if capped {
2010                            break;
2011                        }
2012                        // Periodically persist partial output so it's available if the
2013                        // process is interrupted.
2014                        // Flush eagerly when we have some new content (>256 bytes)
2015                        if assistant_acc.len().saturating_sub(last_flush) > 256 {
2016                            #[cfg(feature = "debug_io")]
2017                            {
2018                                if let Err(e) = std::fs::write(
2019                                    "target/ollama_assistant_acc.txt",
2020                                    &assistant_acc,
2021                                ) {
2022                                    debug!("[ollama] partial write failed: {}", e);
2023                                } else {
2024                                    last_flush = assistant_acc.len();
2025                                    debug!(
2026                                        "[ollama] flushed {} bytes of assistant_acc to disk",
2027                                        last_flush
2028                                    );
2029                                }
2030                            }
2031                        }
2032
2033                        // If we've accumulated something that looks like a JSON object, try to extract it
2034                        if let Some(obj) = extract_json_object(&assistant_acc) {
2035                            if self.early_exit_on_json {
2036                                let ttfb_ms = first_token_at
2037                                    .map(|ft| ft.duration_since(start).as_millis())
2038                                    .unwrap_or(0);
2039                                let total_ms = start.elapsed().as_millis();
2040                                debug!(
2041                                    "[ollama] timing stream: ttfb={}ms, total={}ms (early-exit)",
2042                                    ttfb_ms, total_ms
2043                                );
2044                                return finish_capped(obj);
2045                            }
2046                            text = obj;
2047                            break;
2048                        }
2049                    }
2050                    Err(e) => {
2051                        body_error = Some(anyhow::Error::new(e).context("streaming body error"));
2052                        break;
2053                    }
2054                }
2055
2056                if done_flag {
2057                    break;
2058                }
2059            }
2060            // A lifetime deadline or an idle timeout after at least one token
2061            // returns those tokens. The error is returned only when the
2062            // stream produced none. A UTF-8 or cap error follows the same
2063            // rule: tokens already stored stay.
2064            if text.trim().is_empty() && assistant_acc.trim().is_empty() {
2065                if let Some(err) = body_error {
2066                    return Err(err);
2067                }
2068            }
2069
2070            // Ensure we always flush a final snapshot of assembled content even if small
2071            if !assistant_acc.is_empty() && assistant_acc.len() > last_flush {

2073                {

2075                }
2076            }
2077

2089                        }
2090                    }
2091

2093                    debug!(
2094                        "[ollama] assembled assistant_acc length = {}",
2095                        candidate.len()
2096                    );

2098                        debug!(
2099                            "[ollama] assembled snippet (start): {}",
2100                            &candidate[..200.min(candidate.len())]
2101                        );
2102                        debug!(
2103                            "[ollama] assembled snippet (end): {}",
2104                            &candidate[candidate.len().saturating_sub(200)..]
2105                        );
2106                    } else {
2107                        debug!("[ollama] assembled content: {}", candidate);
2108                    }
2109                    #[cfg(feature = "debug_io")]
2110                    {
2111                        if let Err(e) =
2112                            std::fs::write("target/ollama_assistant_acc.txt", &candidate)

2115                        } else {
2116                            debug!("[ollama] wrote assembled assistant_acc to target/ollama_assistant_acc.txt");
2117                        }

2125                            let ttfb_ms = first_token_at
2126                                .map(|ft| ft.duration_since(start).as_millis())
2127                                .unwrap_or(0);

2133                    } else {

2135                    }
2136                } else if let Ok(s) = std::str::from_utf8(&raw) {
2137                    if let Some(obj) = extract_json_object(s) {
2138                        if self.early_exit_on_json {
2139                            let ttfb_ms = first_token_at
2140                                .map(|ft| ft.duration_since(start).as_millis())
2141                                .unwrap_or(0);

2143                            debug!(
2144                                "[ollama] timing stream: ttfb={}ms, total={}ms (early-exit buffer)",
2145                                ttfb_ms, total_ms
2146                            );
2147                            return finish_capped(obj);
2148                        }
2149                        text = obj;
2150                    } else {
2151                        text = strip_code_fences(s);
2152                    }
2153                }
2154            }
2155
2156            // If we’re here, we didn’t early-exit; log final timings for stream path
2157            let ttfb_ms = first_token_at
2158                .map(|ft| ft.duration_since(start).as_millis())
2159                .unwrap_or(0);
2160            let total_ms = start.elapsed().as_millis();
2161            debug!(
2162                "[ollama] timing stream: ttfb={}ms, total={}ms",
2163                ttfb_ms, total_ms
2164            );

2168            }
2169        }
2170

2172
2173        // Try parsing as JSON value to extract common shapes
2174        if let Ok(v) = serde_json::from_str::<serde_json::Value>(&text) {

2176            if let Some(resp) = v.get("response").and_then(|r| r.as_str()) {

2179
2180            // 2) { "message": { "content": "..." } }
2181            if let Some(msg) = v.get("message") {

2184                    // try to extract and return that JSON directly to simplify downstream parsing.
2185                    if let Some(obj) = extract_json_object(content) {
2186                        return finish_capped(obj);
2187                    }
2188                    let stripped = strip_code_fences(content);
2189                    return finish_capped(stripped);
2190                }
2191            }
2192
2193            // 3) { "choices": [ { "message": { "content": "..." } } ] }
2194            if let Some(choices) = v.get("choices").and_then(|c| c.as_array()) {
2195                if let Some(first) = choices.first() {
2196                    if let Some(msg) = first.get("message") {
2197                        if let Some(content) = msg.get("content").and_then(|c| c.as_str()) {
2198                            return finish_capped(content.to_string());
2199                        }
2200                    }
2201                }
2202            }
2203        }
2204
2205        // If no recognized JSON shape matched, return the raw text as a last resort.
2206        // This covers a body with no newline, which never reached `append_capped`.
2207        finish_capped(text)
2208    }
2209}

2296/// `http` or `https` URL whose host is loopback: `127.0.0.0/8`, `::1`, or `localhost`.
2297///
2298/// Userinfo is rejected, including an empty `@`. This is the only host check.
2299/// Socket binds use `IpAddr::is_loopback` after `SocketAddr` parsing, which is
2300/// the same `127.0.0.0/8` and `::1` rule this function applies to a parsed host.
2301pub fn is_loopback_http_url(url: &str) -> bool {
2302    let Ok(parsed) = url::Url::parse(url) else {
2303        return false;
2304    };
2305    if parsed.scheme() != "http" && parsed.scheme() != "https" {
2306        return false;
2307    }
2308    if url_has_userinfo(url, &parsed) {
2309        return false;
2310    }
2311    match parsed.host() {
2312        Some(url::Host::Domain(domain)) => is_localhost_domain(domain),
2313        Some(url::Host::Ipv4(ip)) => ip.is_loopback(),
2314        Some(url::Host::Ipv6(ip)) => ip.is_loopback(),
2315        None => false,
2316    }
2317}

2322/// `http` or `https` use [`loopback_model_url`] or [`is_loopback_http_url`].

2324    let Ok(parsed) = url::Url::parse(url) else {
2325        return false;
2326    };

2330/// `http://127.0.0.1:11434@evil.example/` puts the loopback text in the userinfo
2331/// and sends the request to `evil.example`. An empty `@` is userinfo too, even
2332/// when the parser drops it and keeps a loopback host.
2333fn url_has_userinfo(raw: &str, parsed: &url::Url) -> bool {
2334    if !parsed.username().is_empty() || parsed.password().is_some() {
2335        return true;
2336    }
2337    let Some(scheme_sep) = raw.find("://") else {
2338        return true;
2339    };
2340    let rest = &raw[scheme_sep + 3..];
2341    let authority = rest.split(['/', '?', '#', '\\']).next().unwrap_or("");
2342    authority.contains('@')
2343}

2347    bare.eq_ignore_ascii_case("localhost") && domain.len() - bare.len() <= 1

2459/// Parse and validate that the produced steps are in the allowed registry (structural check).
2460pub fn parse_llm_plan(json_text: &str, reg: &ToolRegistry) -> Result<PlanIntent> {
2461    // Try direct parse first
2462    if let Ok(plan) = serde_json::from_str::<PlanIntent>(json_text.trim()) {
2463        validate_plan(&plan, reg)?;
2464        return Ok(plan);
2465    }
2466
2467    // Strip common code fences and try again
2468    let cleaned = strip_code_fences(json_text);
2469    // If there's fenced JSON like ```json { ... } ``` try to extract inner JSON first
2470    if let Some(fenced) = extract_json_from_fenced(json_text) {
2471        if let Ok(plan) = serde_json::from_str::<PlanIntent>(fenced.trim()) {
2472            validate_plan(&plan, reg)?;
2473            return Ok(plan);
2474        }
2475        // try cleaned fenced
2476        if let Some(inner_clean) = extract_json_from_fenced(&cleaned) {
2477            if let Ok(plan) = serde_json::from_str::<PlanIntent>(inner_clean.trim()) {
2478                validate_plan(&plan, reg)?;
2479                return Ok(plan);
2480            }
2481        }
2482    }
2483    if let Ok(plan) = serde_json::from_str::<PlanIntent>(cleaned.as_str()) {
2484        validate_plan(&plan, reg)?;
2485        return Ok(plan);
2486    }
2487
2488    // Attempt to extract the last JSON object from the text and parse it
2489    if let Some(obj) = extract_last_json_object(&cleaned) {
2490        if let Ok(plan) = serde_json::from_str::<PlanIntent>(obj.trim()) {
2491            validate_plan(&plan, reg)?;
2492            return Ok(plan);
2493        }
2494    }
2495
2496    // Attempt to extract the first JSON object from the text and parse it
2497    if let Some(obj) = extract_json_object(&cleaned) {
2498        if let Ok(plan) = serde_json::from_str::<PlanIntent>(obj.trim()) {
2499            validate_plan(&plan, reg)?;
2500            return Ok(plan);
2501        }
2502    }
2503
2504    // Try to obtain a JSON Value from several candidates for tolerant extraction
2505    let v_opt: Option<serde_json::Value> =
2506        serde_json::from_str::<serde_json::Value>(cleaned.as_str())
2507            .ok()
2508            .or_else(|| {
2509                extract_last_json_object(json_text).and_then(|s| serde_json::from_str(&s).ok())
2510            })
2511            .or_else(|| {
2512                extract_last_json_object(&cleaned).and_then(|s| serde_json::from_str(&s).ok())
2513            })
2514            .or_else(|| extract_json_object(&cleaned).and_then(|s| serde_json::from_str(&s).ok()));
2515
2516    if let Some(v) = &v_opt {
2517        // Try to locate a nested JSON inside `message.content` or `response`
2518        if let Some(msg) = v.get("message") {
2519            if let Some(content) = msg.get("content").and_then(|c| c.as_str()) {
2520                // Try to parse content as JSON directly
2521                if let Ok(plan) = serde_json::from_str::<PlanIntent>(content.trim()) {
2522                    validate_plan(&plan, reg)?;
2523                    return Ok(plan);
2524                }
2525                // Try to extract JSON from the content string
2526                if let Some(obj2) = extract_json_object(content) {
2527                    if let Ok(plan) = serde_json::from_str::<PlanIntent>(obj2.trim()) {
2528                        validate_plan(&plan, reg)?;
2529                        return Ok(plan);
2530                    }
2531                }
2532            }
2533        }
2534
2535        if let Some(resp_txt) = v.get("response").and_then(|r| r.as_str()) {
2536            if let Some(obj2) = extract_json_object(resp_txt) {
2537                if let Ok(plan) = serde_json::from_str::<PlanIntent>(obj2.trim()) {
2538                    validate_plan(&plan, reg)?;
2539                    return Ok(plan);
2540                }
2541            }
2542        }
2543    }
2544
2545    // Coerce tolerant PlanIntent: accept alternative keys and ensure steps exist
2546    if let Some(v) = v_opt {
2547        let plan_id = (|| {
2548            // common accepted keys
2549            let candidates = [
2550                "plan_id",
2551                "plan_eid",
2552                "id",
2553                "plan_no",
2554                "plan_num",
2555                "planNumber",
2556                "plan_n°",
2557                "plan_n",
2558            ];
2559            for &k in &candidates {
2560                if let Some(vv) = v.get(k) {
2561                    if let Some(s) = vv.as_str() {
2562                        return Some(s.to_string());
2563                    }
2564                }
2565            }
2566
2567            // Try to find any key that, when normalized, matches "planid" or similar
2568            if let Some(obj) = v.as_object() {
2569                for (k, vv) in obj.iter() {
2570                    let norm: String = k
2571                        .chars()
2572                        .filter(|c| c.is_alphanumeric())
2573                        .collect::<String>()
2574                        .to_lowercase();
2575                    if norm == "planid"
2576                        || norm == "plann"
2577                        || norm == "planno"
2578                        || norm == "plannumber"
2579                    {
2580                        if let Some(s) = vv.as_str() {
2581                            return Some(s.to_string());
2582                        }
2583                    }
2584                }
2585            }
2586
2587            None
2588        })();
2589
2590        let plan_id = match plan_id {
2591            Some(id) => id,
2592            None => return Err(anyhow::anyhow!("No valid plan_id found in LLM response")),
2593        };
2594
2595        let steps_val = v
2596            .get("steps")
2597            .cloned()
2598            .unwrap_or(serde_json::Value::Array(vec![]));
2599        let steps_json = serde_json::to_string(&steps_val)?;
2600        let steps: Vec<ActionStep> = serde_json::from_str(&steps_json)?;
2601
2602        let plan = PlanIntent { plan_id, steps };
2603        validate_plan(&plan, reg)?;
2604        return Ok(plan);
2605    }
2606
2607    Err(anyhow::anyhow!(
2608        "Failed to parse LLM plan from text (no valid JSON object found)"
2609    ))
2610}

2623fn validate_plan(plan: &PlanIntent, reg: &ToolRegistry) -> Result<()> {
2624    for s in &plan.steps {
2625        match s {
2626            ActionStep::MoveTo { .. } => {
2627                if !reg.tools.iter().any(|t| t.name == "MoveTo") {
2628                    bail!("LLM used disallowed tool MoveTo");
2629                }
2630            }
2631            ActionStep::Throw { .. } => {
2632                if !reg.tools.iter().any(|t| t.name == "Throw") {
2633                    bail!("LLM used disallowed tool Throw");
2634                }
2635            }
2636            ActionStep::CoverFire { .. } => {
2637                if !reg.tools.iter().any(|t| t.name == "CoverFire") {

2650                    bail!("LLM used disallowed tool {name}");

2652            }
2653        }
2654    }

2667/// Sanitize and validate plan for safety
2668pub fn sanitize_plan(
2669    plan: &mut PlanIntent,
2670    snap: &WorldSnapshot,
2671    reg: &ToolRegistry,
2672) -> Result<()> {
2673    // Remove any steps that exceed bounds or use invalid targets
2674    plan.steps.retain(|step| match step {
2675        ActionStep::MoveTo { x, y, speed: _ } => {
2676            // Check bounds (example: within 100 units)
2677            coord_within_bound(*x)
2678                && coord_within_bound(*y)
2679                && reg.tools.iter().any(|t| t.name == "MoveTo")
2680        }
2681        ActionStep::Throw { item, x, y } => {
2682            // Check item is allowed
2683            matches!(item.as_str(), "smoke" | "grenade")
2684                && coord_within_bound(*x)
2685                && coord_within_bound(*y)
2686                && reg.tools.iter().any(|t| t.name == "Throw")
2687        }
2688        ActionStep::CoverFire {
2689            target_id,
2690            duration,
2691        } => {
2692            // Check target exists and duration reasonable
2693            snap.enemies.iter().any(|e| e.id == *target_id)
2694                && *duration > 0.0
2695                && *duration <= 10.0
2696                && reg.tools.iter().any(|t| t.name == "CoverFire")
2697        }
2698        ActionStep::Revive { ally_id: _ } => {
2699            // Check ally exists (simplified: allow any ally for now, or validate against known ally IDs)
2700            reg.tools.iter().any(|t| t.name == "Revive")
2701        }
2702        // Unknown actions stay out unless the registry names them.
2703        _ => reg.tools.iter().any(|t| t.name == step.action_name()),
2704    });
2705    Ok(())
2706}

2708/// Attempt to extract the first JSON object from a text blob by finding a balanced
2709/// '{' ... '}' region, respecting string literals and escapes.
2710fn extract_json_object(s: &str) -> Option<String> {
2711    let mut depth = 0;
2712    let mut in_string = false;
2713    let mut escape = false;
2714    let mut start = None;
2715
2716    for (i, c) in s.char_indices() {
2717        if in_string {
2718            if escape {
2719                escape = false;
2720            } else if c == '\\' {
2721                escape = true;
2722            } else if c == '"' {
2723                in_string = false;
2724            }
2725        } else {
2726            match c {
2727                '{' => {
2728                    if depth == 0 {
2729                        start = Some(i);
2730                    }
2731                    depth += 1;
2732                }
2733                '}' => {
2734                    if depth > 0 {
2735                        depth -= 1;
2736                        if depth == 0 {
2737                            if let Some(start_idx) = start {
2738                                return Some(s[start_idx..=i].to_string());
2739                            }
2740                        }
2741                    }
2742                }
2743                '"' => in_string = true,
2744                _ => {}
2745            }
2746        }
2747    }
2748    None
2749}

2751/// Extract the last balanced JSON object in a string, if any.
2752fn extract_last_json_object(s: &str) -> Option<String> {
2753    let mut depth = 0;
2754    let mut in_string = false;
2755    let mut escape = false;
2756    let mut start = None;
2757    let mut last_match = None;
2758
2759    for (i, c) in s.char_indices() {
2760        if in_string {
2761            if escape {
2762                escape = false;
2763            } else if c == '\\' {
2764                escape = true;
2765            } else if c == '"' {
2766                in_string = false;
2767            }
2768        } else {
2769            match c {
2770                '{' => {
2771                    if depth == 0 {
2772                        start = Some(i);
2773                    }
2774                    depth += 1;
2775                }
2776                '}' => {
2777                    if depth > 0 {
2778                        depth -= 1;
2779                        if depth == 0 {
2780                            if let Some(start_idx) = start {
2781                                last_match = Some(s[start_idx..=i].to_string());
2782                                // Reset start to find next object
2783                                start = None;
2784                            }
2785                        }
2786                    }
2787                }
2788                '"' => in_string = true,
2789                _ => {}
2790            }
2791        }
2792    }
2793    last_match
2794}

2796/// Find JSON inside fenced code blocks (```json ... ``` or ``` ... ```) and return inner content
2797fn extract_json_from_fenced(s: &str) -> Option<String> {
2798    // look for ```json first
2799    if let Some(start) = s.find("```json") {
2800        if let Some(end_rel) = s[start + 7..].find("```") {
2801            let inner = &s[start + 7..start + 7 + end_rel];
2802            return Some(inner.trim().to_string());
2803        }
2804    }
2805    // fallback to any ``` block
2806    if let Some(start) = s.find("```") {
2807        if let Some(end_rel) = s[start + 3..].find("```") {
2808            let inner = &s[start + 3..start + 3 + end_rel];
2809            return Some(inner.trim().to_string());
2810        }
2811    }
2812    None

2816///
2817/// This function uses a 4-tier fallback chain:
2818/// 1. Full LLM with all 37 tools
2819/// 2. Simplified LLM with 10 most common tools
2820/// 3. Heuristic rule-based planning
2821/// 4. Emergency safe default (Scan + Wait)
2822///
2823/// Cache integration is preserved for performance.
2824pub async fn plan_from_llm(
2825    client: &dyn LlmClient,
2826    snap: &WorldSnapshot,
2827    reg: &ToolRegistry,
2828) -> PlanSource {
2829    #[cfg(feature = "llm_cache")]
2830    {
2831        // Build cache key for fast lookup
2832        let prompt = build_prompt(snap, reg);
2833        let tool_names: Vec<&str> = reg.tools.iter().map(|t| t.name.as_str()).collect();
2834        let cache_key = PromptKey::new(
2835            &prompt,
2836            "default", // TODO: Extract from client
2837            0.7,       // TODO: Extract temperature from client
2838            &tool_names,
2839        );
2840
2841        // Check cache first
2842        if let Some((cached_plan, _decision)) = GLOBAL_CACHE.get(&cache_key) {
2843            debug!(
2844                "[plan_from_llm] Cache HIT - returning cached plan: {}",
2845                cached_plan.plan.plan_id
2846            );
2847
2848            return PlanSource::Llm(cached_plan.plan);
2849        }
2850
2851        debug!("[plan_from_llm] Cache MISS - calling fallback orchestrator");
2852    }
2853
2854    // Cache miss or disabled - use Phase 7 multi-tier fallback
2855    use crate::fallback_system::FallbackOrchestrator;
2856
2857    let orchestrator = FallbackOrchestrator::new();
2858    let result = orchestrator.plan_with_fallback(client, snap, reg).await;
2859
2860    info!(
2861        "[plan_from_llm] Fallback orchestrator succeeded at tier {} after {} attempts ({} ms)",
2862        result.tier.as_str(),
2863        result.attempts.len(),
2864        result.total_duration_ms
2865    );
2866
2867    // Cache successful LLM plans (Tier 1 or Tier 2)
2868    #[cfg(feature = "llm_cache")]
2869    {
2870        if result.tier <= fallback_system::FallbackTier::SimplifiedLlm {

2873            let cache_key = PromptKey::new(&prompt, "default", 0.7, &tool_names);

2894                    "Used {} tier after {} attempts",
2895                    result.tier.as_str(),
2896                    result.attempts.len()
2897                ),
2898            }
2899        }
2900    }
2901}
