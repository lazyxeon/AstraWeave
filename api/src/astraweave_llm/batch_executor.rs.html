1//! Batch Inference Executor
2//!
3//! Enables processing multiple agents in a single LLM call to amortize latency
4//! across agents. This is critical for scalability in multi-agent scenarios.
5//!
6//! # Architecture
7//!
8//! ```text
9//! Agent 1 + Snapshot 1 ┐
10//! Agent 2 + Snapshot 2 ├─→ Batch Prompt (all agents) ─→ LLM (single call)
11//! Agent 3 + Snapshot 3 ┘                                     ↓
12//!                                                   JSON Array: [Plan1, Plan2, Plan3]
13//!                                                             ↓
14//!                                           Distribute: Agent1←Plan1, Agent2←Plan2, Agent3←Plan3
15//! ```
16//!
17//! # Performance
18//!
19//! - **Single agent**: 1.6-2.1s per plan (compressed prompts)
20//! - **Batch of 5**: ~2-3s total = **0.4-0.6s per agent** (4-5× faster)
21//! - **Batch of 10**: ~3-4s total = **0.3-0.4s per agent** (5-7× faster)
22//!
23//! # Determinism
24//!
25//! - Agents processed in deterministic order (sorted by ID)
26//! - Same input order → same plan order
27//! - Critical for replay and multiplayer
28

38/// Agent identifier (must be unique within batch)
39pub type AgentId = u32;

41/// Batch request: multiple agents with their snapshots
42#[derive(Debug, Clone)]
43pub struct BatchRequest {
44    /// Agents to plan for (sorted by ID for determinism)
45    pub agents: Vec<(AgentId, WorldSnapshot)>,
46}
47
48impl BatchRequest {
49    /// Create new batch request
50    pub fn new(agents: Vec<(AgentId, WorldSnapshot)>) -> Self {
51        let mut sorted_agents = agents;
52        // CRITICAL: Sort by ID for determinism
53        sorted_agents.sort_by_key(|(id, _)| *id);
54
55        Self {
56            agents: sorted_agents,
57        }
58    }
59
60    /// Add agent to batch
61    pub fn add_agent(&mut self, id: AgentId, snapshot: WorldSnapshot) {
62        self.agents.push((id, snapshot));
63        // Re-sort to maintain determinism
64        self.agents.sort_by_key(|(id, _)| *id);
65    }
66
67    /// Get batch size
68    pub fn size(&self) -> usize {
69        self.agents.len()
70    }
71
72    /// Check if batch is empty
73    pub fn is_empty(&self) -> bool {
74        self.agents.is_empty()
75    }
76}

78/// Batch response: plans mapped to agent IDs
79#[derive(Debug, Clone)]
80pub struct BatchResponse {
81    /// Plans indexed by agent ID
82    pub plans: HashMap<AgentId, PlanIntent>,
83}
84
85impl BatchResponse {
86    /// Create new batch response
87    pub fn new() -> Self {
88        Self {
89            plans: HashMap::new(),
90        }
91    }
92
93    /// Add plan for agent
94    pub fn add_plan(&mut self, id: AgentId, plan: PlanIntent) {
95        self.plans.insert(id, plan);
96    }
97
98    /// Get plan for agent
99    pub fn get_plan(&self, id: AgentId) -> Option<&PlanIntent> {
100        self.plans.get(&id)
101    }
102
103    /// Get number of plans
104    pub fn len(&self) -> usize {
105        self.plans.len()
106    }
107
108    /// Check if response is empty
109    pub fn is_empty(&self) -> bool {
110        self.plans.is_empty()
111    }
112}

116        Self::new()

120/// Batch prompt builder
121pub struct BatchPromptBuilder;
122
123impl BatchPromptBuilder {
124    /// Build prompt for batch of agents
125    ///
126    /// # Format
127    /// ```text
128    /// You are planning for N agents. Generate EXACTLY N plans in JSON array format.
129    ///
130    /// Agents:
131    /// 1. Agent 1: {snapshot}
132    /// 2. Agent 2: {snapshot}
133    /// ...
134    ///
135    /// Return JSON:
136    /// [
137    ///   {"agent_id": 1, "plan_id": "p1", "steps": [...]},
138    ///   {"agent_id": 2, "plan_id": "p2", "steps": [...]},
139    ///   ...
140    /// ]
141    /// ```
142    pub fn build_batch_prompt(request: &BatchRequest, tool_list: &str) -> String {
143        let n = request.agents.len();
144
145        let mut prompt = format!(
146            r#"You are planning for {} agents. Generate EXACTLY {} plans in JSON array format.
147
148CRITICAL RULES:
149- Return a JSON ARRAY with {} elements
150- Each element MUST have "agent_id", "plan_id", "steps"
151- agent_id MUST match the agent number (1, 2, 3, ...)
152- Use ONLY these tools: {}
153
154"#,
155            n, n, n, tool_list
156        );
157
158        // Add agent snapshots
159        prompt.push_str("Agents:\n");
160        for (idx, (agent_id, snapshot)) in request.agents.iter().enumerate() {
161            let agent_num = idx + 1; // 1-indexed for LLM
162
163            // Use compact JSON for snapshot
164            let snap_json = serde_json::to_string(snapshot).unwrap_or_else(|_| "{}".to_string());
165
166            prompt.push_str(&format!(
167                "{}. Agent {} (ID {}): {}\n",
168                agent_num, agent_num, agent_id, snap_json
169            ));
170        }
171
172        // Add output schema
173        prompt.push_str(
174            r#"
175Return ONLY JSON array (no markdown, no commentary):
176[
177  {"agent_id": 1, "plan_id": "batch-p1", "steps": [{"act": "MoveTo", "x": 10, "y": 5}, ...]},
178  {"agent_id": 2, "plan_id": "batch-p2", "steps": [...]},
179  ...
180]
181"#,
182        );
183
184        prompt
185    }
186}

188/// Batch response parser
189#[derive(Debug, Deserialize, Serialize)]
190struct BatchPlanEntry {
191    agent_id: u32,
192    plan_id: String,
193    steps: Vec<ActionStep>, // Directly deserialize to ActionStep
194}
195
196pub struct BatchResponseParser;
197
198impl BatchResponseParser {
199    /// Parse LLM response into batch response
200    pub fn parse_batch_response(json_text: &str, request: &BatchRequest) -> Result<BatchResponse> {
201        // Try direct parse as array
202        let entries: Vec<BatchPlanEntry> = serde_json::from_str(json_text.trim())
203            .context("Failed to parse batch response as JSON array")?;
204
205        // Validate we got correct number of plans
206        if entries.len() != request.agents.len() {
207            bail!(
208                "Batch response has {} plans but request had {} agents",
209                entries.len(),
210                request.agents.len()
211            );
212        }
213
214        let mut response = BatchResponse::new();
215
216        // Map plans to agent IDs
217        for entry in entries {
218            // Find agent by 1-indexed position (LLM uses 1,2,3...)
219            let agent_idx = (entry.agent_id as usize).saturating_sub(1);
220
221            if agent_idx >= request.agents.len() {
222                bail!(
223                    "Invalid agent_id {} in batch response (max: {})",
224                    entry.agent_id,
225                    request.agents.len()
226                );
227            }
228
229            let (agent_id, _) = request.agents[agent_idx];
230
231            // Steps are already ActionStep objects from deserialization
232            let step_count = entry.steps.len();
233            let plan = PlanIntent {
234                plan_id: entry.plan_id.clone(),
235                steps: entry.steps,
236            };
237
238            response.add_plan(agent_id, plan);
239
240            debug!(
241                "Parsed plan for agent {} (ID {}): {} steps",
242                entry.agent_id, agent_id, step_count
243            );
244        }
245
246        Ok(response)
247    }
248}

250/// Batch inference executor
251pub struct BatchInferenceExecutor {
252    /// Maximum batch size (default: 10)
253    max_batch_size: usize,
254
255    /// Current batch being accumulated
256    current_batch: Option<BatchRequest>,
257}
258
259impl BatchInferenceExecutor {
260    /// Create new batch executor
261    pub fn new() -> Self {
262        Self {
263            max_batch_size: 10,
264            current_batch: None,
265        }
266    }
267
268    /// Create with custom max batch size
269    pub fn with_max_batch_size(max_batch_size: usize) -> Self {
270        Self {
271            max_batch_size,
272            current_batch: None,
273        }
274    }
275
276    /// Queue agent for batch processing
277    pub fn queue_agent(&mut self, id: AgentId, snapshot: WorldSnapshot) {
278        match &mut self.current_batch {
279            Some(batch) => {
280                batch.add_agent(id, snapshot);
281            }
282            None => {
283                let mut batch = BatchRequest::new(Vec::new());
284                batch.add_agent(id, snapshot);
285                self.current_batch = Some(batch);
286            }
287        }
288    }
289
290    /// Check if batch is ready (reached max size or has agents)
291    pub fn is_ready(&self) -> bool {
292        self.current_batch
293            .as_ref()
294            .map(|b| b.size() >= self.max_batch_size)
295            .unwrap_or(false)
296    }
297
298    /// Get current batch size
299    pub fn batch_size(&self) -> usize {
300        self.current_batch.as_ref().map(|b| b.size()).unwrap_or(0)
301    }
302
303    /// Flush current batch (return and clear)
304    pub fn flush_batch(&mut self) -> Option<BatchRequest> {
305        self.current_batch.take()
306    }

308    /// Execute batch inference with streaming support
309    ///
310    /// # Arguments
311    /// * `llm_client` - LLM client (supports streaming for progressive results)
312    /// * `tool_list` - Tool vocabulary string (e.g., "MoveTo|Attack|Reload")
313    ///
314    /// # Returns
315    /// BatchResponse with plans mapped to agent IDs
316    ///
317    /// # Performance
318    /// - Uses streaming to reduce time-to-first-plan
319    /// - Accumulates chunks until full response received
320    /// - Parses complete JSON array at once (deterministic)
321    ///
322    /// # Example
323    /// ```ignore
324    /// use astraweave_llm::{BatchInferenceExecutor, LlmClient, MockLlm};
325    ///
326    /// async fn example() -> anyhow::Result<()> {
327    ///     let mut executor = BatchInferenceExecutor::new();
328    ///     executor.queue_agent(1, snapshot1);
329    ///     executor.queue_agent(2, snapshot2);
330    ///
331    ///     let llm_client: &dyn LlmClient = &MockLlm;
332    ///     let response = executor.execute_batch(llm_client, "MoveTo|Attack").await?;
333    ///     Ok(())
334    /// }
335    /// ```
336    pub async fn execute_batch(
337        &mut self,
338        llm_client: &dyn LlmClient,
339        tool_list: &str,
340    ) -> Result<BatchResponse> {
341        self.execute_batch_formatted(llm_client, tool_list, None)
342            .await
343    }
344
345    /// Same as [`execute_batch`](Self::execute_batch), with an Ollama `format` schema.
346    pub async fn execute_batch_formatted(
347        &mut self,
348        llm_client: &dyn LlmClient,
349        tool_list: &str,
350        format: Option<&serde_json::Value>,
351    ) -> Result<BatchResponse> {
352        let batch = self.flush_batch().context("No batch to execute")?;
353
354        // Build batch prompt
355        let prompt = BatchPromptBuilder::build_batch_prompt(&batch, tool_list);
356
357        debug!(
358            "Executing batch of {} agents (prompt: {} chars)",
359            batch.size(),
360            prompt.len()
361        );
362
363        if let Some(schema) = format {
364            let accumulated = llm_client
365                .complete_with_format(&prompt, schema)
366                .await
367                .context("Failed to complete formatted batch request")?;
368            return BatchResponseParser::parse_batch_response(&accumulated, &batch)
369                .context("Failed to parse batch response");
370        }
371
372        // Stream LLM response and accumulate chunks
373        let mut stream = llm_client
374            .complete_streaming(&prompt)
375            .await
376            .context("Failed to start streaming LLM request")?;
377
378        let mut accumulated = String::new();
379        let mut chunk_count = 0;
380
381        while let Some(chunk_result) = stream.next().await {
382            let chunk = match chunk_result {
383                Ok(chunk) => chunk,
384                Err(err) => {
385                    if accumulated.is_empty() {
386                        return Err(err.context("Failed to receive streaming chunk"));
387                    }
388                    match BatchResponseParser::parse_batch_response(&accumulated, &batch) {
389                        Ok(response) => return Ok(response),
390                        Err(parse_err) => {
391                            let shown = crate::bounded_model_text(&accumulated);
392                            return Err(parse_err.context(format!(
393                                "streaming chunk failed after partial text {shown}: {err:#}"
394                            )));
395                        }
396                    }
397                }
398            };
399
400            let next = accumulated.len().saturating_add(chunk.len());
401            if next > crate::MAX_MODEL_JSON_BYTES {
402                let room = crate::MAX_MODEL_JSON_BYTES.saturating_sub(accumulated.len());
403                let mut end = room.min(chunk.len());
404                while end > 0 && !chunk.is_char_boundary(end) {
405                    end -= 1;
406                }
407                accumulated.push_str(&chunk[..end]);
408                bail!(
409                    "batch text exceeds 64 KiB cap ({next} bytes): {}",
410                    crate::bounded_model_text(&accumulated)
411                );
412            }
413            accumulated.push_str(&chunk);
414            chunk_count += 1;
415
416            debug!(
417                "Received chunk #{}: {} chars (total: {})",
418                chunk_count,
419                chunk.len(),
420                accumulated.len()
421            );
422        }
423
424        debug!(
425            "Streaming complete: {} chunks, {} total chars",
426            chunk_count,
427            accumulated.len()
428        );
429
430        // Parse accumulated response
431        let response = BatchResponseParser::parse_batch_response(&accumulated, &batch)
432            .context("Failed to parse batch response")?;
433
434        debug!(
435            "Batch inference complete: {} plans generated",
436            response.len()
437        );
438
439        Ok(response)
440    }
441}

693                    response[chunk_size * 2..].to_string(),

695            } else {
696                vec![response]
697            };
698
699            Ok(Box::pin(futures_util::stream::iter(
700                chunks.into_iter().map(Ok),
701            )))
702        }

864    struct ChunkThenErr {
865        first: String,
866    }
867
868    impl ChunkThenErr {
869        fn complete(first: String) -> Self {
870            Self { first }
871        }
872
873        fn partial(first: &str) -> Self {
874            Self {
875                first: first.to_string(),
876            }
877        }
878    }

880    #[async_trait::async_trait]
881    impl crate::LlmClient for ChunkThenErr {
882        async fn complete(&self, _prompt: &str) -> Result<String> {
883            Ok(self.first.clone())
884        }
885
886        async fn complete_streaming(
887            &self,
888            _prompt: &str,
889        ) -> Result<std::pin::Pin<Box<dyn futures_util::Stream<Item = Result<String>> + Send>>>
890        {
891            let first = self.first.clone();
892            Ok(Box::pin(futures_util::stream::iter([
893                Ok(first),
894                Err(anyhow::anyhow!("stream idle timeout")),
895            ])))
896        }
897    }
