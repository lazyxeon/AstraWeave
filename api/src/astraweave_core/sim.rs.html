5/// Crafting item ids. Mixed into [`SimRng`] with the world seed and tick.
6pub const SIM_SYSTEM_CRAFT: u64 = 0xC2AF_7001;
7/// Harvest respawn amounts.
8pub const SIM_SYSTEM_HARVEST: u64 = 0xC2AF_7002;
9/// Weaving anchor particle spawns.
10pub const SIM_SYSTEM_ANCHOR_PARTICLES: u64 = 0xAE11_7001;
11/// Enemy patrol retargets. Weaving duplicates this constant; it does not depend on core.
12pub const SIM_SYSTEM_ENEMY_PATROL: u64 = 0xAE11_7002;
13/// Enemy wave spawn position and archetype. Weaving duplicates this constant.
14pub const SIM_SYSTEM_SPAWNER: u64 = 0xAE11_7003;
15/// Terrain solver distance rolls. The solver stores the mixed `u64`.
16pub const SIM_SYSTEM_TERRAIN_SOLVER: u64 = 0x7E22_7001;

18/// Mix `(world_seed, tick, system_id)` into one `StdRng` seed.
19///
20/// Integer-only. The same triple produces the same `u64` on every OS and
21/// architecture. This is not `DefaultHasher` (that algorithm is not stable
22/// across rustc versions).
23pub fn mix_sim_seed(world_seed: u64, tick: u64, system_id: u64) -> u64 {
24    let mut z = world_seed
25        .wrapping_add(0x9E37_79B9_7F4A_7C15)
26        .wrapping_mul(tick ^ 0xBF58_476D_1CE4_E5B9)
27        .wrapping_add(system_id.rotate_left(17));
28    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
29    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
30    z ^ (z >> 31)
31}

33/// Simulation RNG. One stream per `(world_seed, tick, system_id)`.
34///
35/// Draws are `StdRng` (ChaCha12). Reseed from [`SimRng::new`] at the start of
36/// a tick so the sequence does not depend on how many values the previous tick
37/// consumed. Float draws that must be bit-identical across platforms should
38/// use [`SimRng::unit_f32`], which is a dyadic rational, not libm.
39pub struct SimRng {
40    inner: StdRng,
41    world_seed: u64,
42    tick: u64,
43    system_id: u64,
44}
45
46impl SimRng {
47    pub fn new(world_seed: u64, tick: u64, system_id: u64) -> Self {
48        let mixed = mix_sim_seed(world_seed, tick, system_id);
49        Self {
50            inner: StdRng::seed_from_u64(mixed),
51            world_seed,
52            tick,
53            system_id,
54        }
55    }
56
57    pub fn world_seed(&self) -> u64 {
58        self.world_seed
59    }
60
61    pub fn tick(&self) -> u64 {
62        self.tick
63    }
64
65    pub fn system_id(&self) -> u64 {
66        self.system_id
67    }
68
69    /// `f32` in `[0, 1)` from the top 24 bits. Every value is a multiple of
70    /// `2^-24`, so the bit pattern does not depend on libm or on `rand`'s
71    /// float distribution.
72    pub fn unit_f32(&mut self) -> f32 {
73        let u = self.next_u32() >> 8;
74        (u as f32) * (1.0 / 16_777_216.0)
75    }
76
77    /// `f32` in `[-1, 1)`.
78    pub fn signed_unit_f32(&mut self) -> f32 {
79        self.unit_f32() * 2.0 - 1.0
80    }
81}

83impl RngCore for SimRng {
84    fn next_u32(&mut self) -> u32 {
85        self.inner.next_u32()
86    }
87
88    fn next_u64(&mut self) -> u64 {
89        self.inner.next_u64()
90    }
91
92    fn fill_bytes(&mut self, dest: &mut [u8]) {
93        self.inner.fill_bytes(dest)
94    }
95}

152        let mut a = SimRng::new(0xA11C, 7, SIM_SYSTEM_CRAFT);
153        let mut b = SimRng::new(0xA11C, 7, SIM_SYSTEM_CRAFT);
